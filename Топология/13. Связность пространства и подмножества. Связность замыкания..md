## Определение

$$
\Large
\text{Пространство } (X,\Omega)\text{ несвязно, если }
\exists\, U_1,U_2\in\Omega_X :
$$

$$
\Large
U_1,U_2\neq\varnothing
\;;\;
U_1\cup U_2=X
\;;\;
U_1\cap U_2=\varnothing
$$

$$
\Large
\Longleftrightarrow
\exists\, U\neq\varnothing, X :
\;
U \text{ открыто и замкнуто одновременно}
$$

$$
\Large
(\text{потому что: } U_1:=U,\; U_2:=X\setminus U)
$$

## Определение
$$
\Large
 A\subset X,\;
A \text{ несвязное подмножество } X, \text{ если}
$$
$$\Large
 A \text{ несвязно как ТП}
$$

$$
\Large
\text{с индуцированной топологией.}
$$
---

$$
\Large
\widetilde{U}_1 \subset A \text{ открыто}
\Longleftrightarrow
\exists\, U_1\in\Omega_X :
\widetilde{U}_1=U_1\cap A
$$

$$
\Large
A \text{ несвязно}
\Longleftrightarrow
\exists\, \widetilde{U}_1,\widetilde{U}_2
\text{ открытые в } A,\text{ разбивающие } A
$$

$$
\Large
\Longleftrightarrow
\exists\, U_1,U_2 \text{ открытые в } X :
\widetilde{U}_i=U_i\cap A
$$

$$
\Large
\widetilde{U}_1 \text{ и } \widetilde{U}_2
\text{ разбивают } A:
$$

$$
\Large
\widetilde{U}_i\neq\varnothing
\Rightarrow
U_i\cap A\neq\varnothing
$$

$$
\Large
\widetilde{U}_1\cup\widetilde{U}_2=A
\Rightarrow
(U_1\cup U_2)\supset A
$$

$$
\Large
\widetilde{U}_1\cap\widetilde{U}_2=\varnothing
\Rightarrow
U_1\cap U_2\cap A=\varnothing
$$
$$\Large
\text{Теперь перефразируем определение:}

## Определение
$$
\Large
 A\subset X,\;
A \text{ несвязно, если }
\exists\, U_1,U_2 \text{ открытые в } X :
$$

$$
\Large
U_1\cap A\neq\varnothing
\;;\;
U_2\cap A\neq\varnothing
$$

$$
\Large
U_1\cap U_2\cap A=\varnothing
$$

$$
\Large
U_1\cup U_2\supset A
$$
---
## Примеры
$$
\Large
1)\;
X \text{ — антидискретное пространство}
$$

$$
\Large
\forall\, A\subset X
\;\;
A \text{ связно}
$$

$$
\Large
2)\;
X \text{ — дискретное пространство}
\qquad
A\subset X
$$

$$
\Large
|A|>1
\;\Rightarrow\;
A \text{ несвязно}
$$

$$
\Large
|A|=1
\;\Rightarrow\;
A \text{ связно всегда}
$$

$$
\Large
U_1=\{a_0\},
\qquad
U_2=A\setminus U_1
$$

$$
\Large
3)\;
X \text{ — топология Зарисского}
\qquad
A\subset X
$$

$$
\Large
\Rightarrow
A \text{ — тоже топология Зарисского}
$$

$$
\Large
1<|A|<\infty
\;\Rightarrow\;
A \text{ несвязно}
\qquad
(\text{на } A \text{ дискретная топология})
$$

$$
\Large
|A|=\infty
\;\Rightarrow\;
A \text{ связно}
$$

$$
\Large
4)\;
X \text{ — стрелка}
\qquad

$$

$$
\Large
\text{открытые множества } (a,+\infty)
$$

$$
\Large
A\subset X
\qquad
\forall\, A
\text{ — связно}
$$

## Теорема
  
$$  
\Large  
\text{Если } X \text{ — топологическое пространство, }  
A \text{ — связно, }  
A\subset X,  
\;  
A\subset B\subset \operatorname{Cl}A,  
$$  
  
$$  
\Large  
\text{то } B \text{ связно.}  
$$  
## Доказательство
$$
\Large
\text{Допустим } B \text{ несвязно.}
$$

$$
\Large
\text{Пусть }\ U_1,U_2\in\Omega_X :
$$

$$
\Large
U_1\cap B\neq\varnothing
\;;\;
U_2\cap B\neq\varnothing
\;;\;
U_1\cap U_2\cap B=\varnothing
\;;\;
U_1\cup U_2\supset B
$$

$$
\Large
B\supset A
\Rightarrow
$$

$$
\Large
U_1\cap U_2\cap A=\varnothing
\;;\;
U_1\cup U_2\supset A
$$

$$
\Large
\text{НУО: } 
U_1\cap A=\varnothing
$$

$$
\Large
\Rightarrow
U_2\supset A
$$

$$
\Large
\operatorname{Cl}A\setminus U_1
\text{ замкнуто}
$$

$$
\Large
\text{и содержит } A,
\text{ так как } U_1\cap A=\varnothing
$$

$$
\Large
\operatorname{Cl}A\setminus U_1\neq \operatorname{Cl}A
$$

$$
\Large
\text{так как }
B\cap U_1\neq\varnothing
$$

$$
\Large
(
\exists\, x\in B\cap U_1 :
x\in \operatorname{Cl}A,
\;
x\notin \operatorname{Cl}A\setminus U_1
)
$$

$$
\Large
\text{Противоречие: получили замкнутое множество,}
$$

$$
\Large
\text{меньшее чем } \operatorname{Cl}A,
\text{ но содержащее } A.
$$

---
## Следствие
  
$$  
\Large  
A \text{ связно}  
\;\Rightarrow\;  
\operatorname{Cl}A \text{ связно.}  
$$
