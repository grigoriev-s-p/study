## Теорема

$$
\Large
[0,1]
\text{ компактно.}
$$

## Доказательство

$$
\Large
\text{Допустим противное.}
$$

$$
\Large
\{U_i\}_{i\in I}
\text{ — открытое покрытие в }\mathbb R,
\qquad
\bigcup_{i\in I}U_i\supset[0,1]
$$

$$
\Large
\text{и нет конечного подпокрытия.}
$$

$$
\Large
[0,\tfrac12]
\text{ и }
[\tfrac12,1]
$$

$$
\Large
\text{— один из них не покрывается конечным числом }U_i.
$$

$$
\Large
\text{Разобьём его на две половины и т.д.}
$$

$$
\Large
[0,1]\supset[a_1,b_1]\supset[a_2,b_2]\supset\dots
$$

$$
\Large
|b_k-a_k|
=
\frac12|b_{k-1}-a_{k-1}| \xrightarrow[k \to \infty]{ } 0
$$

$$
\Large
\Rightarrow
\text{по лемме о вложенных отрезках}
$$

$$
\Large
\bigcap_{i=1}^{\infty}[a_i,b_i]=\{x_0\}



$$

$$
\Large
x_0\in U_{i_0}
\quad
(\exists U_{i_0})
$$

$$
\Large
\Rightarrow
\exists\varepsilon>0:
\quad
B(x_0,\varepsilon)\subset U_{i_0}
$$
$$
\Large
\text{Возьмём } n, \text{ т.ч. }
\frac1{2^n}<\varepsilon
$$

$$
\Large
[a_n,b_n]\subset B(x_0,\varepsilon)\subset U_{i_0}
$$

$$
\Large
\text{Но по построению }
[a_n,b_n]
\text{ не покрывается конечным числом }U_i
$$

$$
\Large
\text{Противоречие}
$$

## Лемма Лебега

$$
\Large
\{U_i\}
\text{ — открытое покрытие }
[0,1]
$$

$$
\Large
\Rightarrow
\exists\varepsilon>0
$$

$$
\Large
(\varepsilon
\text{ называется числом Лебега покрытия }
\{U_i\})
$$

$$
\Large
\forall x_0\in[0,1]
\quad
B(x_0,\varepsilon)\subset U_i
\text{ для некоторого }U_i.
$$

## Доказательство

$$
\Large
\text{Допустим противное: }
$$

$$
\Large
\varepsilon_i=\frac1{2^i}
$$

$$
\Large
\text{и ни одно из них не подходит.}
$$

$$
\Large
\forall i
\quad
\exists x_i:
\quad
B(x_i,\varepsilon_i)\not\subset U_j
\quad
\forall j.
$$

$$
\Large
\Rightarrow
\text{Можем выделить сходящуюся подпоследовательность: }
\exists x_{i_k}\to x_0
$$

$$
\Large
\exists j
\text{ т.ч. }
x_0\in U_j
\Rightarrow
\exists\varepsilon>0:
\quad
B(x_0,\varepsilon)\subset U_j.
$$

$$
\Large
x_{i_k}\in B\left(x_0,\frac{\varepsilon}{2}\right)\subset U_j
$$

$$
\Large
\text{Выберем }k
\text{ так, что }
\varepsilon_{i_k}<\frac{\varepsilon}{2}.
$$

$$
\Large
B(x_{i_k},\varepsilon_{i_k})
\subset
B(x_0,\varepsilon)
\subset
U_j
$$

$$
\Large
\text{противоречие.}
$$
