$$
\Large
 A \subset M - \text{ метр. пр-во},\ \ x_0 \in M
$$
1)
$$
\Large
x_0 \text{ называют внутренней точкой для } A, 
$$
$$ \Large
\text{если } \exists \varepsilon > 0:\ B(x_0,\varepsilon)\subset A \quad (x_0 \in A)
$$
2)
$$
\Large
x_0 \text{ называют внешней точкой для } A 
$$
$$ \Large
\text{если } \exists \varepsilon > 0:\ B(x_0,\varepsilon)\cap A=\varnothing \quad (x_0 \notin A)
$$
3)
$$
\Large
 x_0 \text{ — граничная точка, если:}
$$
$$
\Large
\forall \varepsilon>0:\ B(x_0,\varepsilon)\not\subset A \ \ \text{и} \ \ B(x_0,\varepsilon)\cap A\ne\varnothing
$$

$$
\Large
\text{Докажем, что } \ B(x_1, \varepsilon) \subset B(x_0, r)

\quad

## Теорема(свойства открытых множеств)
1.
$$  
\Large  
\{U_i\}_{i\in I}\ \text{— открытые}  
\;\Rightarrow\;  
\bigcup_{i\in I} U_i  
\ \text{— открыто}  
$$  
2.
$$  
\Large  
U_1,U_2,\dots,U_n  
\ \text{— открытые(конечное кол-во)}  
\;\Rightarrow\;  
\bigcap_{i=1}^{n} U_i  
\ \text{— открыто}  
$$  
3.
$$  
\Large  
\varnothing  
\ \text{и}\  
M  
\ \text{— открытые}  
$$  
  
## Доказательство  
1.  
$$  
\Large  
x_0 \in \bigcup_{i\in I} U_i  
\;\Rightarrow\;  
x_0 \in U_i  
\ (\exists i)  
$$  
  
$$  
\Large  
\Rightarrow  
\exists \varepsilon>0:  
\quad  
B(x_0,\varepsilon)\subset U_i  
\subset  
\bigcup_{i\in I} U_i  
$$  
  
2.
$$  
\Large  
x_0 \in \bigcap_{i=1}^{n} U_i  
\;\Rightarrow\;  
\forall i  
\quad  
x_0\in U_i  
$$  
  
$$  
\Large  
\Rightarrow  
\exists \varepsilon_i>0:  
\quad  
B(x_0,\varepsilon_i)\subset U_i  
$$  
  
$$  
\Large  
\varepsilon=\min\{\varepsilon_i\}  
$$  
  
$$  
\Large  
B(x_0,\varepsilon)  
\subset  
B(x_0,\varepsilon_i)  
\subset  
U_i  
$$  
  
$$  
\Large  
\Rightarrow  
B(x_0,\varepsilon)  
\subset  
\bigcap_{i=1}^{n} U_i  
$$

## Теорема(Равносильность определений внутренности)
1)
$$\Large
\text{Множество внутренних точек A} := B
$$
2)
$$\Large
\text{Объединение всех открытых множеств}
$$
$$\Large
\bigcup  U :=C\qquad U \text{ - открытое},\ U \subset A 
$$
3)
$$\Large
\text{максимальное открытое множество, содержащееся в } A
$$
## Доказательство
$$\Large
(2)\iff(3)\ \text{очев.}
$$

$$\Large
\text{Докажем: } \ (1)\iff(2)
$$

$$\Large
\text{Хотим: } B=C
$$
1)
$$\Large
\text{Докажем } \ B \subset C
$$
$$\Large
\forall x_0\in B \quad x_0 \text{ - внутр. точка } A
$$

$$\Large
\text{(Для любой внутренней точки существует открытое множество содержашее её)}
$$

$$\Large
\exists U_{x_0} \subset A, \quad U_{x_0} \text{ - открытое, такое что: } \ x_0 \in  U_{x_0}\subset C
$$
$$\Large
B\subset C \ \text{(т.к. для любой точки $B$ найдется открытое множество,}
$$
$$
\Large
\text{ по которым и строится объединение $C$)}
$$
2)
$$\Large
\text{Докажем } \ C \subset B
$$
$$\Large
\forall x_0\in C \qquad C \text{ - открытое},\quad C\subset A
\Rightarrow
x_0 \text{ - внутренняя точка } A \Rightarrow x_0\in B
$$
$$\Large
\Rightarrow C \subset B 
$$

$$\Large
\Rightarrow B = C \qquad \text{чтд}
$$
