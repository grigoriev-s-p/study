## Теорема

$$
\Large
\text{Пусть } A_1, A_2 \text{ - связны, }
A_1\cap A_2\neq\varnothing,
$$

$$
\Large
\Rightarrow A_1\cup A_2 \text{ связно.}
$$

## Доказательство

$$
\Large
\text{Допустим } A_1\cup A_2 \text{ несвязно.}
$$

$$
\Large
\Rightarrow \exists\, U_1,U_2 :
$$

$$
\Large
\varnothing
=
U_i\cap(A_1\cup A_2)
=
(U_i\cap A_1)\cup(U_i\cap A_2)
\qquad (i=1,2)
$$
$$
\Large
U_1\cap U_2\cap(A_1\cup A_2)=\varnothing \ \Rightarrow \ U_1\cap U_2\cap A_1=\varnothing
$$

$$
\Large
(U_1\cup U_2) \supset (A_1\cup A_2) \ \Rightarrow \ U_1\cup U_2\supset A_1
$$

$$
\Large
x_0\in A_1\cap A_2
$$


$$
\Large
\text{Не умаляя общности }
U_1\cap A_1=\varnothing
$$

$$
\Large
x_0\in U_2
$$

$$
\Large
\exists\, x_1\in U_1\cap A_2
$$

$$
\Large
(
U_1\cap(A_1\cup A_2)\neq\varnothing, \text{ но }
\
U_1\cap A_1=\varnothing
)
$$

$$
\Large
\Rightarrow
U_1\cap A_2\neq\varnothing
\qquad (x_1)
$$

$$
\Large
U_2\cap A_2\neq\varnothing
\qquad (x_0)
$$

$$
\Large
U_1\cap U_2\cap A_2=\varnothing
$$

$$
\Large
U_1\cup U_2\supset A_2
$$

$$
\Large
\Rightarrow
A_2 \text{ несвязно}
$$

$$
\Large
\text{Противоречие}
$$

## Теорема

$$
\Large
f:X\to Y \text{ непрерывно},\qquad A\subset X \text{ связно}
\;\Rightarrow\;
f(A) \text{ связно.}
$$

## Доказательство

$$
\Large
\text{Предположим, что } f(A) \text{ несвязно.}
$$

$$
\Large
\Rightarrow
\exists\, U_1,U_2 \subset Y - \text{ открыты,}
$$

$$
\Large
U_1\cap f(A)\neq\varnothing,\qquad
U_2\cap f(A)\neq\varnothing,
$$

$$
\Large
U_1\cap U_2\cap f(A)=\varnothing,
\qquad
U_1\cup U_2\supset f(A).
$$

$$
\Large
\text{Так как } f \text{ непрерывно, то }
f^{-1}(U_1),\,f^{-1}(U_2)
\text{ открыты в } X.
$$

$$
\Large
\text{Покажем, что } f^{-1}(U_1),\,f^{-1}(U_2)
\text{ разбивают } A.
$$

$$
\Large
U_1\cap f(A)\neq\varnothing
\ \Leftrightarrow
\,f^{-1}(U_1)\cap A\neq\varnothing:
$$

$$
\Large
\exists\, y\in U_1\cap f(A)
\;\Rightarrow\;
\exists\, x\in A:\ f(x)=y,
$$

$$
\Large
\Rightarrow
x\in f^{-1}(U_1)\cap A.
$$

$$
\Large
\text{Аналогично }
f^{-1}(U_2)\cap A\neq\varnothing.
$$

$$
\Large
f^{-1}(U_1)\cap f^{-1}(U_2)\cap A
=
f^{-1}(U_1\cap U_2)\cap A
=
\varnothing,
$$

$$
\Large
\text{поскольку } U_1\cap U_2\cap f(A)=\varnothing.
$$

$$
\Large
\text{Далее, если } x\in A,\text{ то } f(x)\in f(A)\subset U_1\cup U_2,
$$

$$
\Large
\text{значит } x\in f^{-1}(U_1)\cup f^{-1}(U_2).
$$

$$
\Large
\Rightarrow
A\subset f^{-1}(U_1)\cup f^{-1}(U_2).
$$

$$
\Large
\text{Итак, } f^{-1}(U_1),\,f^{-1}(U_2)
\text{ разделяют } A.
$$

$$
\Large
\Rightarrow
A \text{ несвязно,}
$$

$$
\Large
\text{что противоречит условию.}
$$

$$
\Large
\Rightarrow
f(A)\text{ связно.}
$$
