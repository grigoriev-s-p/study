# Дифференцируемость функции одной переменной

$$
\Large
f:\mathbb R\to\mathbb R\text{ дифференцируема в }x\iff\exists A=\lim_{h\to0}\frac{f(x+h)-f(x)}h\in\mathbb R
$$

$$
\Large
\iff\exists A\in\mathbb R:\quad A=\frac{f(x+h)-f(x)}h+\overline o(1),\quad h\to0
$$

$$
\Large
\iff\exists A\in\mathbb R:\quad f(x+h)=f(x)+Ah+\overline o(h),\quad h\to0
$$

---

# Определение. Дифференцируемость отображения и дифференциал

$$
\Large
\text{Пусть }f:D\subset\mathbb R^n\to\mathbb R^m,\qquad x\text{ — внутренняя точка }D\quad(x\in\operatorname{int}D)
$$

$$
\Large
f\text{ дифференцируема в точке }x,\text{ если }\exists L\in\mathcal L(\mathbb R^n,\mathbb R^m):
$$

$$
\Large
f(x+h)=f(x)+Lh+\varphi(h),\qquad\varphi(h)=\overline o(\|h\|)\text{ при }h\to\Theta_n
$$

$$
\Large
\text{То есть }\frac{\|\varphi(h)\|}{\|h\|}\xrightarrow[h\to\Theta_n]{}0,\qquad\Theta_n=(0,\ldots,0)
$$

$$
\Large
\text{Обозначение }\overline o\text{ означает «о малое»}
$$

$$
\Large
L\text{ называется дифференциалом, полной производной или производным оператором функции }f\text{ в точке }x
$$

---

# Обозначения. Дифференциал и его действие на приращение

$$
\Large
f'(x),\qquad d_xf,\qquad df(x)
$$

$$
\Large
Lh=f'(x)h=d_xf(h)=df(x,h)
$$

---

# Определение. Матрица Якоби

$$
\Large
\text{Матрица дифференциала называется матрицей Якоби и также обозначается }f'(x)
$$

---

# Свойство 1. Дифференцируемость влечёт непрерывность

$$
\Large
f\text{ дифференцируема в }x\implies f\text{ непрерывна в }x
$$

### Доказательство

$$
\Large
f(x+h)-f(x)=Lh+\overline o(\|h\|)\xrightarrow[h\to0]{}\Theta_m
$$

---

# Свойство 2. Представление остаточного слагаемого

$$
\Large
\varphi(h)=\overline o(h),\quad h\to\Theta_n\iff\varphi(h)=\alpha(h)\|h\|
$$

$$
\Large
\text{Где }\alpha(\Theta_n)=\Theta_m\text{ и }\alpha\text{ непрерывна в }\Theta_n
$$

---

# Свойство 3. Дифференциал постоянной функции

$$
\Large
f:\mathbb R^n\to\mathbb R^m\text{ постоянна},\qquad f(x)=c\in\mathbb R^m\quad\forall x\in\mathbb R^n
$$

### Доказательство

$$
\Large
f(x+h)=f(x)+0_{m\times n}h+0\implies d_xf\text{ — нулевой линейный оператор}
$$

$$
\Large
f'(x)=
\begin{pmatrix}
0&\cdots&0\\
\vdots&\ddots&\vdots\\
0&\cdots&0
\end{pmatrix}
$$

---

# Свойство 4. Дифференциал аффинной функции

$$
\Large
f:\mathbb R^n\to\mathbb R^m,\qquad f(x)=Ax+B,\qquad A\text{ — матрица размера }m\times n,\quad B\in\mathbb R^m
$$

### Доказательство

$$
\Large
f(x+h)=f(x)+Ah+\Theta_m\implies f'(x)=A
$$

---

# Лемма 1. Единственность дифференциала

$$
\Large
\text{Дифференциал единственен}
$$

### Доказательство

$$
\Large
\text{Пусть }f:D\subset\mathbb R^n\to\mathbb R^m\text{ дифференцируема в }x\in\operatorname{int}D
$$

$$
\Large
\implies f(x+h)=f(x)+Lh+\overline o(\|h\|),\quad h\to\Theta_n,\quad\text{для некоторого }L\in\mathcal L(\mathbb R^n,\mathbb R^m)
$$

$$
\Large
\text{Зафиксируем }h\in\mathbb R^n\setminus\{\Theta_n\}
$$

$$
\Large
\text{Рассмотрим }f(x+th)=f(x)+tLh+\overline o(t),\qquad t\to0,\quad t\in\mathbb R
$$

$$
\Large
\implies Lh=\frac{f(x+th)-f(x)}t+\overline o(1),\quad t\to0
$$

$$
\Large
\implies Lh=\lim_{t\to0}\frac{f(x+th)-f(x)}t
$$

$$
\Large
\text{В силу единственности предела значение }Lh\text{ однозначно определено пределом}
$$

---

### Пример. Матрица Якоби функции со значениями в R

$$
\Large
\text{Пусть }f:D\subset\mathbb R^n\to\mathbb R\text{ дифференцируема в точке }x
$$

$$
\Large
\text{Матрица Якоби }f'(x)\text{ — вектор в }\mathbb R^n,\qquad f'(x)=(\alpha_1,\ldots,\alpha_n)
$$

$$
\Large
d_xf(h)=(\alpha_1,\ldots,\alpha_n)\begin{pmatrix}h_1\\\vdots\\h_n\end{pmatrix}=\langle\alpha,h\rangle=\sum_{k=1}^n\alpha_kh_k
$$

---

# Определение. Градиент функции

$$
\Large
f:D\subset\mathbb R^n\to\mathbb R\text{ дифференцируема в точке }x,\text{ если }\exists\alpha\in\mathbb R^n:
$$

$$
\Large
f(x+h)=f(x)+\langle\alpha,h\rangle+\overline o(\|h\|),\qquad h\to\Theta_n
$$

$$
\Large
\text{Вектор }\alpha\text{ называется градиентом функции }f\text{ в }x
$$

$$
\Large
\alpha=\operatorname{grad}f(x)=\nabla_xf=\nabla f(x)
$$

$$
\Large
\text{Символ }\nabla\text{ — «набла»}
$$

# Лемма 2. Дифференцируемость отображения и его координатных функций

$$
\Large
f:D\subset\mathbb R^n\to\mathbb R^m,\qquad f=(f_1,\ldots,f_m)
$$

$$
\Large
f\text{ дифференцируема в точке }x\iff\forall k=1,\ldots,m\quad f_k\text{ дифференцируема в }x
$$

$$
\Large
\text{Строки матрицы Якоби }f\text{ совпадают с }\operatorname{grad}f_k
$$

$$
\Large
f'(x)=
\begin{pmatrix}
\operatorname{grad}f_1(x)\\
\operatorname{grad}f_2(x)\\
\vdots\\
\operatorname{grad}f_m(x)
\end{pmatrix}
$$

### Доказательство

$$
\Large
(\Rightarrow)\quad f\text{ дифференцируема в }x\iff\exists L\in\mathcal L(\mathbb R^n,\mathbb R^m)
$$

$$
\Large
f(x+h)=f(x)+Lh+\overline o(h)\tag{*}
$$

$$
\Large
L:\mathbb R^n\to\mathbb R^m,\qquad L=(L_1,\ldots,L_m)^T,\qquad L_k:\mathbb R^n\to\mathbb R
$$

$$
\Large
f_k(x+h)=f_k(x)+L_kh+\varphi_k(h),\qquad h\to\Theta_n\tag{**}
$$

$$
\Large
\varphi_k(h)\text{ — }k\text{-я координата остатка }\overline o(h)\text{ в }(*)
$$

$$
\Large
(\Leftarrow)\quad\text{Берём }L=(L_1,\ldots,L_m)^T,\text{ где }L_k\text{ удовлетворяет }(**)
$$
