# Определение. Изолированная точка множества

$$
\Large
\text{Пусть }(X,d)\text{ — метрическое пространство},\qquad D\subset X
$$

$$
\Large
p\in D\text{ — изолированная точка, если }\exists V_p:\quad V_p\cap D=\{p\}
$$

---

# Замечание. Неизолированная точка множества

$$
\Large
\text{Если }p\in D\text{ не изолированная, то }p\text{ — предельная точка }D
$$

---

# Определение. Непрерывность отображения в точке

$$
\Large
\text{Пусть }(X,d_x),\ (Y,d_y)\text{ — метрические пространства},\quad D\subset X,\quad f:D\to Y,\quad p\in D
$$

$$
\Large
\begin{gathered}
f\text{ непрерывно в точке }p,\text{ если выполнено одно}\\
\text{из следующих эквивалентных условий}
\end{gathered}
$$

$$
\Large
1)\quad p\text{ — изолированная точка }D\quad\text{или}\quad\lim_{x\to p}f(x)=f(p)
$$

$$
\Large
2)\quad\forall\varepsilon>0\ \exists\delta>0:\quad\forall x\in D\quad d_x(x,p)<\delta\implies d_y(f(x),f(p))<\varepsilon
$$

$$
\Large
3)\quad\forall V_{f(p)}\ \exists V_p:\quad f(V_p\cap D)\subset V_{f(p)}
$$

$$
\Large
4)\quad\forall\{x_n\}\subset D\quad x_n\xrightarrow[n\to\infty]{}p\implies f(x_n)\xrightarrow[n\to\infty]{}f(p)
$$

# Теорема. Непрерывность композиции отображений

$$
\Large
(X,d_x),\ (Y,d_y),\ (Z,d_z)\text{ — метрические пространства},\qquad D\subset X,\quad f:D\to Y
$$

$$
\Large
f(D)\subset G,\qquad g:G\to Z
$$

$$
\Large
\begin{gathered}
\text{Пусть }f\text{ непрерывно в }x_0\in D,\quad g\text{ непрерывно в }y_0=f(x_0)\\
\implies g\circ f\text{ непрерывно в }x_0
\end{gathered}
$$

### Доказательство

$$
\Large
\text{Рассмотрим }V_{g(f(x_0))}\text{ — окрестность }g(f(x_0))=g(y_0)
$$

$$
\Large
\implies\exists V_{y_0}=V_{f(x_0)}:\quad g(V_{y_0})\subset V_{g(y_0)}=V_{g(f(x_0))}
$$

$$
\Large
\text{По непрерывности }g\text{ в }y_0\text{ и }f\text{ в }x_0
$$

$$
\Large
\exists V_{x_0}:\quad f(V_{x_0})\subset V_{f(x_0)}\implies g(f(V_{x_0}))\subset g(V_{f(x_0)})=g(V_{y_0})\subset V_{g(y_0)}
$$

$$
\Large
\implies(g\circ f)(V_{x_0})\subset V_{(g\circ f)(x_0)}
$$

# Теорема. Характеристика непрерывности через прообразы открытых множеств

$$
\Large
(X,d_x),\ (Y,d_y)\text{ — метрические пространства},\qquad f:X\to Y
$$

$$
\Large
f\text{ непрерывно на }X\iff\forall G\subset Y\text{ открытого}\quad f^{-1}(G)\text{ открыто}
$$

### Доказательство

$$
\Large
(\implies)\quad\text{Пусть }G\subset Y\text{ открыто},\qquad f:X\to Y\text{ непрерывно на }X
$$

$$
\Large
\forall x_0\in f^{-1}(G)\implies f(x_0)=y_0\in G,\quad G\text{ открыто}\implies\exists V_{y_0}\subset G
$$

$$
\Large
\text{Так как }f\text{ непрерывно в }x_0\implies\exists V_{x_0}:\quad f(V_{x_0})\subset V_{y_0}\subset G\implies V_{x_0}\subset f^{-1}(G)
$$

$$
\Large
\implies x_0\text{ — внутренняя точка }f^{-1}(G)\implies f^{-1}(G)\text{ открыто}
$$

$$
\Large
(\impliedby)\quad\text{Пусть }\forall G\subset Y\text{ открытого}\quad f^{-1}(G)\text{ открыто}
$$

$$
\Large
\text{Рассмотрим }x_0\in X\quad\text{и докажем, что }f\text{ непрерывно в }x_0
$$

$$
\Large
\text{Пусть }V_{f(x_0)}\text{ — окрестность }f(x_0),\quad V_{f(x_0)}\text{ открыто}\implies f^{-1}(V_{f(x_0)})\text{ открыто}
$$

$$
\Large
x_0\in f^{-1}(V_{f(x_0)})\implies\exists V_{x_0}\subset f^{-1}(V_{f(x_0)})\implies f(V_{x_0})\subset V_{f(x_0)}
$$

# Теорема. Непрерывный образ компакта

$$
\Large
\text{Пусть }f:X\to Y,\quad X\text{ — компакт}\implies f(X)\text{ — компакт}
$$

$$
\Large
\text{Здесь }(X,d_x),\ (Y,d_y)\text{ — метрические пространства}
$$

### Доказательство

$$
\Large
\text{Пусть }\{G_\alpha\}_{\alpha\in I}\text{ — открытое покрытие }f(X)
$$

$$
\Large
f(X)\subset\bigcup_{\alpha\in I}G_\alpha,\quad G_\alpha\text{ открыто}\implies f^{-1}(G_\alpha)\text{ открыто}
$$

$$
\Large
X=f^{-1}(f(X))\subset f^{-1}\left(\bigcup_{\alpha\in I}G_\alpha\right)=\bigcup_{\alpha\in I}f^{-1}(G_\alpha)
$$

$$
\Large
\implies\{f^{-1}(G_\alpha)\}_{\alpha\in I}\text{ — открытое покрытие }X
$$

$$
\Large
\implies\exists I_1\subset I\text{ конечное}:\quad X\subset\bigcup_{\alpha\in I_1}f^{-1}(G_\alpha)
$$

$$
\Large
\implies f(X)\subset f\left(\bigcup_{\alpha\in I_1}f^{-1}(G_\alpha)\right)\subset\bigcup_{\alpha\in I}f(f^{-1}(G_\alpha))=\bigcup_{\alpha\in I_1}G_\alpha
$$

# Следствие 1. Образ компактного подмножества

$$
\Large
f:X\to Y\text{ непрерывно на }X,\quad K\subset X\text{ — компакт}\implies f(K)\text{ — компакт}
$$

---

# Следствие 2. Теорема Вейерштрасса для непрерывных отображений

$$
\Large
\text{Пусть }(X,d)\text{ — метрическое пространство},\quad K\subset X\text{ — компакт}
$$

$$
\Large
\begin{gathered}
f:K\to\mathbb R\text{ непрерывно на }K\\
\implies f\text{ ограничено и достигает минимума и максимума на }K
\end{gathered}
$$

### Доказательство

$$
\Large
f(K)\text{ — компакт}\implies f(K)\text{ ограничено и замкнуто}
$$

$$
\Large
\implies f\text{ ограничено}\implies\exists m,M\in\mathbb R:\quad m=\inf_{x\in K}f(x),\quad M=\sup_{x\in K}f(x)
$$

$$
\Large
\text{Докажем, что }m\in f(K)
$$

$$
\Large
\text{Действительно, }\exists x_n\in K\text{ такой, что }m\le f(x_n)\le m+\frac1n
$$

$$
\Large
\implies f(x_n)\xrightarrow[n\to\infty]{}m\implies m\text{ — предельная точка }f(K)\implies m\in f(K)\implies\exists x_0\in K:\quad f(x_0)=m
$$

$$
\Large
\text{Аналогично, }\exists x_1\in K:\quad f(x_1)=M\implies m=\min f,\quad M=\max f
$$

# Определение. Равномерная непрерывность отображения

$$
\Large
f:D\subset X\to Y\text{ равномерно непрерывно на }D,\text{ если}
$$

$$
\Large
\forall\varepsilon>0\ \exists\delta>0:\quad\forall x_1,x_2\in D\quad d_x(x_1,x_2)<\delta\implies d_y(f(x_1),f(x_2))<\varepsilon
$$

---

# Теорема Кантора. Равномерная непрерывность на компакте

$$
\Large
(X,d_x),\ (Y,d_y)\text{ — метрические пространства}
$$

$$
\Large
K\subset X\text{ — компакт},\quad f:K\to Y\text{ непрерывно на }K\implies f\text{ равномерно непрерывно на }K
$$

### Доказательство

$$
\Large
\text{Пусть }f\text{ не является равномерно непрерывным}\implies\exists\varepsilon_0>0
$$

$$
\Large
\exists x_1^{(n)},x_2^{(n)}\in K:\quad d_x(x_1^{(n)},x_2^{(n)})<\frac1n,\quad d_y(f(x_1^{(n)}),f(x_2^{(n)}))\ge\varepsilon_0\quad *
$$

$$
\Large
\exists n_j\nearrow\nearrow:\quad x_1^{(n_j)}\xrightarrow[j\to\infty]{}x\in K
$$

$$
\Large
\text{Так как }d(x_1^{(n)},x_2^{(n)})<\frac1n\implies x_2^{(n_j)}\to x\in K
$$

$$
\Large
f\text{ непрерывно в }x\in K
$$

$$
\Large
\implies\left.\begin{gathered}
f(x_2^{(n_j)})\xrightarrow[j\to\infty]{}f(x)\\
f(x_1^{(n_j)})\xrightarrow[j\to\infty]{}f(x)
\end{gathered}\right\}\implies d(f(x_1^{(n_j)}),f(x_2^{(n_j)}))\xrightarrow[j\to\infty]{*}0\quad\text{(!!!)}
$$
