# Определение. Норма линейного оператора

$$
\Large
\text{Пусть }(X,\|\cdot\|_X),(Y,\|\cdot\|_Y)\text{ — нормированные пространства}
$$
$$
\Large
L:X\to Y\text{ — линейный оператор}
$$

$$
\Large
\text{Нормой линейного оператора }L\text{ называется }\|L\|=\|L\|_{X\to Y}=\sup_{\|x\|_X\le1}\|Lx\|_Y
$$

---

# Определение. Ограниченный линейный оператор

$$
\Large
L:X\to Y\text{ ограничен, если }\|L\|<+\infty
$$

---

# Обозначение. Пространство ограниченных линейных операторов

$$
\Large
\mathcal L(X,Y)\text{ — пространство ограниченных линейных операторов }X\to Y
$$

$$
\Large
\text{Для операторов }X\to X\text{ используется обозначение }\mathcal L(X)
$$

---

# Следствие 2. Эквивалентность ограниченности и непрерывности линейного оператора

$$
\Large
X,Y\text{ — нормированные пространства},\qquad L:X\to Y\text{ — линейный оператор}
$$

$$
\Large
\text{Следующие условия эквивалентны}
$$

$$
\Large
\begin{gathered}
1)\quad L\text{ — ограниченный линейный оператор}\\
2)\quad L\text{ непрерывен в }\Theta_X\\
3)\quad L\text{ непрерывен на }X\\
4)\quad L\text{ равномерно непрерывен}\\
5)\quad E\text{ ограничено}\implies L(E)\text{ ограничено}\\
6)\quad L(B(\Theta_X,1))\text{ ограничено}
\end{gathered}
$$

## Доказательство

$$
\Large
1\Rightarrow2:\quad L\Theta_X=\Theta_Y,\qquad\lim_{x\to\Theta}\|Lx\|=0,\text{ так как }\|Lx\|\le\|L\|\|x\|
$$

$$
\Large
\implies Lx\xrightarrow[x\to\Theta_X]{}\Theta_Y=L\Theta_X\implies L\text{ непрерывен в нуле}
$$

$$
\Large
2\Rightarrow4:\quad x_0\in X,\quad Lx_0=y_0,\qquad\|Lx-Lx_0\|\le\|L\|\|x-x_0\|
$$

$$
\Large
\text{Пусть }\varepsilon>0,\quad\|x-x_0\|<\delta=\frac{\varepsilon}{\|L\|}\implies\|Lx-Lx_0\|\le\|L\|\|x-x_0\|<\varepsilon
$$

$$
\Large
4\Rightarrow3\text{ — очевидно},\qquad3\Rightarrow2\text{ — очевидно}
$$

$$
\Large
2\Rightarrow1:\quad\text{Пусть }L\text{ — линейный оператор, непрерывный в }\Theta_X
$$

$$
\Large
\varepsilon=1,\quad\exists\delta>0:\quad\|x\|=\|x-\Theta_X\|<\delta\implies\|Lx\|=\|Lx-L\Theta_X\|\le1
$$

$$
\Large
\text{Пусть }x\in X\setminus\{\Theta_X\},\qquad\widetilde x=\delta\frac{x}{\|x\|}\implies\|\widetilde x\|=\delta\implies\|L\widetilde x\|\le1
$$

$$
\Large
\|Lx\|\le\frac1\delta\|x\|\implies\|L\|\le\frac1\delta<+\infty\implies L\text{ ограничен}
$$

$$
\Large
5\Rightarrow6\text{ — очевидно}
$$

$$
\Large
1\iff6:\quad L\text{ ограничен}\iff\exists C>0:\quad\forall\|x\|<1\quad\|Lx\|\le C
$$

$$
\Large
\iff\exists C>0:\quad L(B(\Theta_X,1))\subset B(\Theta_Y,C)
$$

$$
\Large
1\Rightarrow5
$$

$$
\Large
\text{Пусть }E\text{ ограничено},\quad\exists R>0:\quad\forall x\in E\quad\|x\|\le R
$$

$$
\Large
\|Lx\|\le\|L\|\|x\|\le\|L\|R\implies Lx\in B(\Theta_Y,\|L\|R)
$$

$$
\Large
\implies LE\subset B(\Theta_Y,\|L\|R)\text{ — ограниченное множество}
$$

# Теорема. Оценка нормы оператора, порождённого матрицей

$$
\Large
\text{Пусть }L_A:\mathbb R^n\to\mathbb R^m\text{ — линейный оператор, порождённый матрицей }A=(a_{jk})_{\substack{j=1,\ldots,m\\k=1,\ldots,n}}
$$

$$
\Large
L_A\text{ ограничен},\qquad\|L_A\|^2\le\sum_{j=1}^m\sum_{k=1}^n a_{jk}^2
$$

## Доказательство

$$
\Large
\text{Пусть }x\in\mathbb R^n
$$

$$
\Large
\|L_Ax\|^2=\left\|\sum_{j=1}^m\left(\sum_{k=1}^n a_{jk}x_k\right)e^j\right\|^2=\sum_{j=1}^m\left(\sum_{k=1}^n a_{jk}x_k\right)^2\overset{\text{Коши — Буняковского — Шварца}}{\le}\left(\sum_{j=1}^m\sum_{k=1}^n a_{jk}^2\right)\|x\|^2
$$
