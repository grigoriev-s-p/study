# Свойство 1. Дифференциал линейной комбинации отображений

$$
\Large
f,g:D\subset\mathbb R^n\to\mathbb R^n\text{ дифференцируемы в }x\in\operatorname{int}D,\qquad\alpha,\beta\in\mathbb R
$$

$$
\Large
\implies\alpha f+\beta g\text{ дифференцируема в }x,\qquad d_x(\alpha f+\beta g)=\alpha d_xf+\beta d_xg
$$

$$
\Large
(\alpha f+\beta g)'(x)=\alpha f'(x)+\beta g'(x)
$$

## Доказательство

$$
\Large
f(x+h)=f(x)+d_xf(h)+\varphi_1(h),\qquad g(x+h)=g(x)+d_xg(h)+\varphi_2(h)
$$

$$
\Large
\varphi_1(h),\varphi_2(h)=\overline o(h),\qquad h\to\Theta_n
$$

$$
\Large
(\alpha f+\beta g)(x+h)=(f+g)+(\alpha d_xf+\beta d_xg)(h)+\underbrace{\bigl(\varphi_1(h)+\varphi_2(h)\bigr)}_{\overline o(h),\ h\to\Theta_n}
$$

$$
\Large
\text{Матрица }\alpha d_xf+\beta d_xg\text{ — }\alpha f'(x)+\beta g'(x)
$$

# Свойство 2. Дифференциал скалярного произведения отображений

$$
\Large
\langle f,g\rangle=\sum_{k=1}^m f_kg_k
$$

$$
\Large
\langle f,g\rangle\text{ дифференцируемо в }x,\qquad d_x\langle f,g\rangle(h)=\langle d_xf(h),g(x)\rangle+\langle f(x),d_xg(h)\rangle
$$

$$
\Large
\langle f,g\rangle'=g^T(x)f'(x)+f^T(x)g'(x),\qquad\underbrace{g^T(x)}_{1\times m}\underbrace{f'(x)}_{m\times n}
$$

$$
\Large
m=1\implies\langle f,g\rangle'=\nabla f(x)\,g(x)+f(x)\,\nabla g(x)
$$

## Доказательство

$$
\Large
m=1
$$

$$
\Large
\langle f,g\rangle(x+h)=\langle f(x)+d_xf(h)+\alpha(h)|h|,\ g(x)+d_xg(h)+\beta(h)|h|\rangle
$$

$$
\Large
\alpha,\beta:\mathbb R^n\to\mathbb R^m,\qquad\alpha\text{ непрерывна в }\Theta_n,\qquad\alpha(\Theta_n)=\beta(\Theta_n)=\Theta_m
$$

$$
\Large
\begin{aligned}
\langle f,g\rangle(x+h)&=\langle f,g\rangle+\langle f(x),d_xg(h)\rangle+\langle d_xf(h),g(x)\rangle\\
&\quad+\underbrace{\langle\alpha(h),g(x+h)\rangle|h|+\langle f(x+h),\beta(h)\rangle|h|+\langle d_xf(h),d_xg(h)\rangle}_{\overline o(h)\text{ — докажем}}
\end{aligned}
$$

$$
\Large
\langle\alpha(h),g(x+h)\rangle\le|\alpha(h)|\,|g(x+h)|\xrightarrow[h\to0]{}0
$$

$$
\Large
\alpha(h)\to\Theta_m,\qquad g(x+h)\text{ ограничено и }g(x+h)\to g(x)
$$

$$
\Large
\text{Второе слагаемое аналогично}
$$

$$
\Large
\frac{\langle d_xf(h),d_xg(h)\rangle}{|h|}\le\underbrace{\frac{|d_xf(h)|}{|h|}}_{\text{ограничено, так как }d_xf\text{ — ограниченный линейный оператор}}\,|d_xg(h)|\le\|d_xf\|\,\|d_xg\|\,|h|\xrightarrow[h\to0]{}0
$$

$$
\Large
\implies\langle f,g\rangle\text{ дифференцируемо в }x
$$

# Свойство 3. Дифференциал произведения скалярной функции и отображения

$$
\Large
f:D\subset\mathbb R^n\to\mathbb R^m,\qquad\lambda:D\subset\mathbb R^n\to\mathbb R
$$

$$
\Large
f,\lambda\text{ дифференцируемы в }x\implies\lambda f\text{ дифференцируемо в }x
$$

$$
\Large
d_x(\lambda f)(h)=d_x\lambda(h)f(x)+\lambda(x)d_xf(h)
$$

$$
\Large
d_x\lambda(h)=\langle\nabla\lambda(x),h\rangle
$$

$$
\Large
(\lambda f)'(x)=\underbrace{f(x)}_{m\times1}\underbrace{\lambda'(x)}_{1\times n}+\underbrace{\lambda(x)}_{1\times1}\underbrace{f'(x)}_{m\times n}
$$

## Доказательство (упражнение)

$$
\Large
L=d_xf,\qquad l=d_x\lambda
$$

$$
\Large
f(x+h)=f(x)+Lh+\varphi(h),\qquad\lambda(x+h)=\lambda(x)+lh+\psi(h),\qquad\varphi(h),\psi(h)=\overline o(h)
$$

$$
\Large
(\lambda f)(x+h)-(\lambda f)(x)=\lambda(x)Lh+(lh)f(x)+R(h)
$$

$$
\Large
R(h)=\lambda(x)\varphi(h)+\psi(h)f(x)+(lh+\psi(h))(Lh+\varphi(h))
$$

$$
\Large
\frac{|R(h)|}{|h|}\le|\lambda(x)|\frac{|\varphi(h)|}{|h|}+|f(x)|\frac{|\psi(h)|}{|h|}+\left(\|l\|+\frac{|\psi(h)|}{|h|}\right)\bigl(\|L\||h|+|\varphi(h)|\bigr)\xrightarrow[h\to\Theta_n]{}0
$$

$$
\Large
h\mapsto\lambda(x)Lh+(lh)f(x)\text{ — линейный оператор}\implies d_x(\lambda f)(h)=\lambda(x)d_xf(h)+d_x\lambda(h)f(x)
$$
