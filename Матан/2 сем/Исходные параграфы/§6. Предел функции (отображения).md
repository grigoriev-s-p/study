# Определение. Предел отображения

$$
\Large
\begin{gathered}
\text{Пусть }(X,d_x),\ (Y,d_y)\text{ — метрические пространства},\quad D\subset X,\\
p\text{ — предельная точка }D
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Пределом отображения }f:D\to Y\text{ в точке }p\\
\text{называется элемент }A\in Y\text{ такой, что выполнено}\\
\text{одно из следующих эквивалентных условий}
\end{gathered}
$$

$$
\Large
1)\quad\text{Условие }\varepsilon\text{–}\delta\text{, или по Коши}
$$

$$
\Large
\forall\varepsilon>0\ \exists\delta>0:\quad\forall x\in D\quad 0<d_x(x,p)<\delta\implies d_y(f(x),A)<\varepsilon
$$

$$
\Large
\begin{gathered}
2)\quad\forall V_A\text{ — окрестность }A\quad\exists\mathring V_p\text{ — проколотая окрестность }p\\
\text{такая, что }f(\mathring V_p\cap D)\subset V_A
\end{gathered}
$$

$$
\Large
3)\quad\text{По Гейне, на языке последовательностей}
$$

$$
\Large
\forall\{x_n\}\subset D\setminus\{p\}\quad x_n\xrightarrow[n\to\infty]{}p\implies f(x_n)\xrightarrow[n\to\infty]{}A
$$

$$
\Large
\text{Обозначения: }A=\lim_{x\to p}f(x),\qquad f(x)\xrightarrow[x\to p]{}A
$$

# Свойство 1. Эквивалентность условий определения предела

$$
\Large
\text{Условия 1–3 определения эквивалентны}
$$

### Доказательство

$$
\Large
1\implies2:\quad\text{Пусть }V_A=B(A,\varepsilon)\text{ — окрестность }A
$$

$$
\Large
\begin{gathered}
\text{По условию 1 }\exists\delta>0:\quad\forall x\in D\\
0<d_x(x,p)<\delta\implies d_y(f(x),A)<\varepsilon
\end{gathered}
$$

$$
\Large
\implies\forall x\in\bigl(D\cap B(p,\delta)\bigr)\setminus\{p\}\quad f(x)\in V_A\implies\exists\mathring V_p=B(p,\delta)\setminus\{p\}:\quad f(D\cap\mathring V_p)\subset V_A
$$
$$
$$
$$
$$
$$
$$
$$
\Large
2\implies3:\quad\text{Пусть }\{x_n\}\subset D\setminus\{p\},\qquad x_n\xrightarrow[n\to\infty]{}p
$$

$$
\Large
\text{Докажем, что }f(x_n)\xrightarrow[n\to\infty]{}A
$$

$$
\Large
\text{Пусть }V_A\text{ — окрестность }A\implies\exists\mathring V_p:\quad f(\mathring V_p\cap D)\subset V_A
$$

$$
\Large
\begin{gathered}
\text{Так как }x_n\to p,\quad x_n\in D\setminus\{p\},\\
\exists N\in\mathbb N:\quad\forall n\ge N\quad x_n\in\mathring V_p\cap D\implies f(x_n)\in V_A\implies f(x_n)\xrightarrow[n\to\infty]{}A
\end{gathered}
$$
$$
$$
$$
$$
$$
$$
$$
\Large
3\implies1:\quad\text{Через противоречие}
$$

$$
\Large
\exists\varepsilon_0>0\ \forall\delta>0\ \exists x\in D:\quad 0<d_x(x,p)<\delta,\quad d_y(f(x),A)>\varepsilon_0
$$

$$
\Large
\forall n\in\mathbb N\ \exists x_n\in D:\quad 0<d_x(x_n,p)<\frac1n,\quad d_y(f(x_n),A)>\varepsilon_0
$$

$$
\Large
\implies\{x_n\}\subset D\setminus\{p\},\quad x_n\xrightarrow[n\to\infty]{}p\quad\text{и}\quad f(x_n)\not\longrightarrow A\quad\text{(3 не выполняется!)}
$$

# Свойство 2. Единственность предела отображения

$$
\Large
\text{Предел отображения единственен}
$$

### Доказательство

$$
\Large
\text{Пусть }A\text{ и }B\text{ удовлетворяют определению предела}
$$

$$
\Large
p\text{ — предельная точка }D\implies\exists\{x_n\}\subset D\setminus\{p\}:\quad x_n\to p
$$

$$
\Large
x_n\to p\implies\begin{cases}f(x_n)\xrightarrow[n\to\infty]{}A\\[4pt]f(x_n)\xrightarrow[n\to\infty]{}B\end{cases}\implies A=B\quad\text{как пределы одной последовательности}
$$

# Свойство 3. Ограниченность отображения в окрестности точки

$$
\Large
\begin{gathered}
\text{Если }f\text{ имеет предел в точке }p,\\
\text{то }f\text{ ограничено в некоторой окрестности }p
\end{gathered}
$$

### Доказательство

$$
\Large
\text{Пусть }V_A\text{ — окрестность предела }A=\lim_{x\to p}f(x)\implies\exists\mathring V_p:\quad f(\mathring V_p\cap D)\subset V_A
$$

$$
\Large
\implies f(V_p\cap D)\subset V_A\cup\{f(p)\}\subset\widetilde V_A
$$

# Свойство 4. Теорема о пределе композиции отображений(?????)

$$
\Large
\text{Пусть }(X,d_x),\ (Y,d_y),\ (Z,d_z)\text{ — метрические пространства}
$$

$$
\Large
D\subset X,\qquad f:D\to Y,\qquad f(D)\subset G,\qquad g:G\to Z
$$

$$
\Large
\text{Предположим, что}
$$

$$
\Large
1)\quad x_0\text{ — предельная точка }D
$$

$$
\Large
2)\quad\exists y_0=\lim_{x\to x_0}f(x)
$$

$$
\Large
3)\quad\exists\mathring V_{x_0}:\quad\forall x\in\mathring V_{x_0}\quad f(x)\ne y_0
$$

$$
\Large
4)\quad z_0=\lim_{y\to y_0}g(y)\implies z_0=\lim_{x\to x_0}(g\circ f)(x)
$$

### Доказательство по Гейне

$$
\Large
\text{Пусть }\{x_n\}_{n=1}^{\infty}\subset D\setminus\{x_0\},\qquad x_n\to x_0
$$

$$
\Large
\implies f(x_n)\xrightarrow[n\to\infty]{}y_0,\quad\text{из условия 3 }\exists N\in\mathbb N:\quad\forall n>N\quad f(x_n)\ne y_0
$$

$$
\Large
\implies y_n=f(x_n)\in G\setminus\{y_0\},\quad n\ge N,\qquad y_n\to y_0
$$

$$
\Large
\implies g(y_n)\xrightarrow[n\to\infty]{}z_0\implies g(f(x_n))\to z_0\implies z_0=\lim_{x\to x_0}(g\circ f)(x)
$$

### Доказательство по определению 1, по Коши (упражнение)

$$
\Large
\text{Пусть }\varepsilon>0
$$

$$
\Large
\lim_{y\to y_0}g(y)=z_0\implies\exists\eta>0:\quad\forall y\in G\quad 0<d_y(y,y_0)<\eta\implies d_z(g(y),z_0)<\varepsilon
$$

$$
\Large
\lim_{x\to x_0}f(x)=y_0\implies\exists\delta_1>0:\quad\forall x\in D\quad 0<d_x(x,x_0)<\delta_1\implies d_y(f(x),y_0)<\eta
$$

$$
\Large
\text{Из условия 3 }\exists r>0:\quad\forall x\in D\quad 0<d_x(x,x_0)<r\implies f(x)\ne y_0
$$

$$
\Large
\text{Пусть }\delta=\min\{\delta_1,r\}>0
$$

$$
\Large
\forall x\in D\quad 0<d_x(x,x_0)<\delta\implies f(x)\in G,\quad 0<d_y(f(x),y_0)<\eta\implies d_z(g(f(x)),z_0)<\varepsilon
$$

$$
\Large
\implies\lim_{x\to x_0}(g\circ f)(x)=z_0
$$

### Доказательство по определению 2, через окрестности (упражнение)

$$
\Large
\text{Пусть }V_{z_0}\text{ — произвольная окрестность }z_0
$$

$$
\Large
\begin{gathered}
\lim_{y\to y_0}g(y)=z_0\implies\exists W_{y_0}\text{ — окрестность }y_0:\\
g(\mathring W_{y_0}\cap G)\subset V_{z_0},\qquad\mathring W_{y_0}=W_{y_0}\setminus\{y_0\}
\end{gathered}
$$

$$
\Large
\lim_{x\to x_0}f(x)=y_0\implies\exists\mathring U_{x_0}:\quad f(\mathring U_{x_0}\cap D)\subset W_{y_0}
$$

$$
\Large
\text{Из условия 3 }\exists\mathring V_{x_0}:\quad\forall x\in\mathring V_{x_0}\cap D\quad f(x)\ne y_0
$$

$$
\Large
\text{Пусть }\mathring U'_{x_0}=\mathring U_{x_0}\cap\mathring V_{x_0}\text{ — проколотая окрестность }x_0
$$

$$
\Large
f(\mathring U'_{x_0}\cap D)\subset\bigl(W_{y_0}\setminus\{y_0\}\bigr)\cap G=\mathring W_{y_0}\cap G
$$

$$
\Large
\implies(g\circ f)(\mathring U'_{x_0}\cap D)=g\bigl(f(\mathring U'_{x_0}\cap D)\bigr)\subset g(\mathring W_{y_0}\cap G)\subset V_{z_0}\implies\lim_{x\to x_0}(g\circ f)(x)=z_0
$$
