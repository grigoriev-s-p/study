# Определение. Открытый шар

$$
\Large
\text{Пусть }(X,d)\text{ — метрическое пространство},\qquad x\in X,\quad r>0
$$

$$
\Large
B(x,r)=\{y\in X:\ d(y,x)<r\}
$$

$$
\Large
\text{Открытый шар с центром }x\text{ и радиусом }r
$$

# Определение. Замкнутый шар

$$
\Large
\overline{B}(x,r)=\{y\in X:\ d(y,x)\le r\}
\text{ — замкнутый шар}
$$

$$
\Large
S(x,r)=\{y\in X:\ d(y,x)=r\}\text{ — сфера}
$$

# Определение. Внутренняя точка

$$
\Large
\text{Пусть }(X,d)\text{ — метрическое пространство},\qquad D\subset X
$$

$$
\Large
\begin{gathered}
 p\in D\text{ — внутренняя точка }D,\text{ если}\\[4pt]
\exists r>0\text{ такой, что }B(p,r)\subset D
\end{gathered}
$$

---

# Определение. Открытое множество

$$
\Large
\begin{gathered}
 D\text{ открыто, если}\\[4pt]
\text{каждая точка }D\text{ — внутренняя}
\end{gathered}
$$

---

# Определение. Внутренность множества

$$
\Large
\begin{gathered}
D^\circ=\operatorname{int}D =\{p\in D:\ p\text{ — внутренняя точка }D\}\\[4pt]
\text{— внутренность}
\end{gathered}
$$

---

# Замечание

$$
\Large
D\text{ открыто}\iff D=D^\circ
$$

# Лемма. Открытый шар — открытое множество

### Доказательство

$$
\Large
\begin{gathered}
\text{Пусть }B(p,r)=\{x\in X:\ d(x,p)<r\}\\[4pt]
\implies\rho=r-d(x,p)>0
\end{gathered}
$$

$$
\Large
\begin{gathered}
B(x,\rho)\subset B(p,r):\\[4pt]
\text{Пусть }y\in B(x,\rho)\implies\\[4pt]
d(y,p)\le d(x,y)+d(x,p) <r-d(x,p)+d(x,p)=r
\end{gathered}
$$

---

### Примеры открытых множеств

$$
\Large
X,\ \varnothing\text{ — открыты}
$$

---

# Теорема. Свойства открытых множеств

$$
\Large
\begin{gathered}
1)\quad\bigcup\text{ любого количества открытых множеств}\\
\text{— открыто}
\end{gathered}
$$

$$
\Large
\begin{gathered}
2)\quad\bigcap\text{ конечного семейства открытых множеств}\\
\text{— открыто}
\end{gathered}
$$

$$
\Large
\begin{gathered}
3)\quad\bigcap\text{ бесконечного семейства открытых множеств}\\
\text{может не быть открытым}
\end{gathered}
$$

$$
\Large
4)\quad D\subset X\implies\operatorname{int}D\text{ открыта}
$$

### Доказательство 

$$
\Large
\begin{gathered}
1)\quad\{D_\alpha\}_{\alpha\in I}\text{ — семейство множеств}\\[4pt]
\forall\alpha\in I:\quad D_\alpha\text{ открыто}
\end{gathered}
$$

$$
\Large
\bigcup_{\alpha\in I}D_\alpha =\{x:\ \exists\alpha\in I\text{ такая, что }x\in D_\alpha\}
$$

$$
\Large
\begin{gathered}
\text{Пусть }x\in\bigcup_{\alpha\in I}D_\alpha \implies\exists\alpha_0\in I:\ x\in D_{\alpha_0}\\[4pt]
\implies\exists r>0:\quad
B(x,r)\subset D_{\alpha_0}\subset\bigcup_{\alpha\in I}D_\alpha
\end{gathered}
$$

---



$$
\Large
2)\quad\text{Пусть }D_1,\ldots,D_m\subset X\text{ — открыты}
$$

$$
\Large
\begin{gathered}
x\in\bigcap_{k=1}^{m}D_k\\[4pt]
\implies\forall k=1,\ldots,m\quad
\exists r_k>0:\ B(x,r_k)\subset D_k
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Пусть }r=\min(r_1,\ldots,r_m)>0\\[4pt]
\implies B(x,r)\subset\bigcap_{k=1}^{m}B(x,r_k)
\subset\bigcap_{k=1}^{m}D_k
\end{gathered}
$$

---



$$
\Large
\begin{gathered}
3)\quad X=\mathbb{R},\qquad
D_k=\left(-\frac{1}{k},\frac{1}{k}\right)\\[4pt]
\implies\bigcap_{k=1}^{\infty}D_k=\{0\}\text{ — не открыто}\\[4pt]
\text{(Евклидова метрика)}
\end{gathered}
$$

$$
\Large
\forall r>0:\quad(-r,r)\not\subset\{0\}
$$

---



$$
\Large
4)\quad\text{Пусть }D\subset X:\quad\operatorname{int}D\text{ открыта}
$$

$$
\Large
\text{Пусть }x\in\operatorname{int}D \implies\exists B(x,r)\subset D
$$

$$
\Large
\begin{gathered}
\forall y\in B(x,r)\\[4pt]
\exists\rho>0:\quad B(y,\rho)\subset B(x,r)\subset D\\[4pt]
\implies y\in\operatorname{int}D \implies B(x,r)\subset\operatorname{int}D\\[4pt]
\implies x\text{ — внутренняя точка }\operatorname{int}D\\[4pt]
\implies\operatorname{int}D\text{ открыта}
\end{gathered}
$$

\text{имеет непустое пересечение с }E:\\[4pt]

# Определение. Замкнутое множество

$$
\Large
\begin{gathered}
E\subset X\text{ замкнуто, если содержит}\\
\text{все свои предельные точки}
\end{gathered}
$$

# Определение. Замыкание множества

$$
\Large
\begin{gathered}
\text{Замыкание множества }E\text{ — }E\cup\text{множество предельных точек }E:\\[4pt]
\operatorname{cl}E=\overline{E} =E\cup\{p:\ p\text{ — предельная точка }E\}
\end{gathered}
$$

---

### Пример

$$
\Large
\begin{gathered}
E=[0,1)\cup\{2\}\subset\mathbb{R}\\[4pt]
\text{с евклидовой метрикой}
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Все точки }[0,1]\text{ — предельные,}\\
\text{а точка }2\text{ не является предельной}\\[4pt]
\operatorname{cl}E=[0,1]\cup\{2\}
\end{gathered}
$$

# Теорема. Замкнутость множества и открытость дополнения

$$
\Large
(X,d)\text{ — метрическое пространство},\qquad E\subset X
$$

$$
\Large
E\text{ замкнуто}\iff X\setminus E\text{ открыто}
$$

### Доказательство

$$
\Large
\begin{gathered}
\quad\text{Пусть }E\text{ замкнуто}\\[4pt]
\text{Пусть }p\in X\setminus E\\[4pt]
\implies p\text{ не является предельной точкой }E\\[4pt]
\implies\exists r>0\text{ такой, что}\\[4pt]
\underbrace{\bigl(B(p,r)\setminus\{p\}\bigr)}_{\mathring{V}_p}
\cap E=\varnothing
\end{gathered}
$$

$$
\Large
\begin{gathered}
p\notin E\implies B(p,r)\cap E=\varnothing\\[4pt]
\implies\underbrace{B(p,r)}_{V_p}\subset X\\[4pt]
\implies p\text{ — внутренняя точка}\\[4pt]
\implies X\setminus E\text{ открыто}
\end{gathered}
$$
$$
$$
$$
$$
$$
$$
$$\LARGE
(\Leftarrow):
$$
$$
\Large
\begin{gathered}
\quad\text{Пусть }X\setminus E\text{ открыто,}\\[4pt]
p\in X\text{ — предельная точка }E\\[4pt]
\text{Предположим, что }p\notin E\\[4pt]
\implies p\in X\setminus E\\[4pt]
\implies\exists V_p:\quad V_p\subset X\setminus E\\[4pt]
\implies\mathring{V}_p\cap E=\varnothing\\[4pt]
\implies p\text{ не является предельной точкой }E\text{ (противоречие)}\\[4pt]
\implies p\in E\\[4pt]
\implies E\text{ содержит все свои предельные точки}
\end{gathered}
$$

# Следствие 1. Замкнутость замкнутого шара

$$
\Large
\overline{B}(p,r)\text{ замкнут}
$$

### Доказательство

$$
\Large
\text{Докажем, что }X\setminus\overline{B}(p,r)\text{ открыто}
$$

$$
\Large
\begin{gathered}
\text{Пусть }y\in X\setminus\overline{B}(p,r),\qquad d(p,y)>r\\[4pt]
\implies\rho=d(p,y)-r>0
\end{gathered}
$$

$$
\Large
\text{Докажем, что }B(y,\rho)\subset X\setminus\overline{B}(p,r)
$$

$$
\Large
\begin{gathered}
\text{Пусть }z\in B(y,\rho)\\[4pt]
d(p,z)\ge d(p,y)-d(y,z) >d(y,p)-\rho =d(y,p)-d(y,p)+r=r\\[4pt]
\implies B(y,\rho)\subset X\setminus\overline{B}(p,r)
\end{gathered}
$$

# Лемма. Формулы де Моргана

$$
\Large
\text{Пусть }\{G_\alpha\}_{\alpha\in I}
$$

$$
\Large
1)\quad X\setminus\left(\bigcup_{\alpha\in I}G_\alpha\right) =\bigcap_{\alpha\in I}(X\setminus G_\alpha)
$$

$$
\Large
2)\quad X\setminus\left(\bigcap_{\alpha\in I}G_\alpha\right) =\bigcup_{\alpha\in I}(X\setminus G_\alpha)
$$

### Доказательство

$$
\Large
\begin{aligned}
1)\quad x\in X\setminus\left(\bigcup_{\alpha\in I}G_\alpha\right) &\iff x\notin\bigcup_{\alpha\in I}G_\alpha\\[4pt]
&\iff\forall\alpha\in I:\ x\notin G_\alpha\\[4pt]
&\iff\forall\alpha\in I:\ x\in X\setminus G_\alpha\\[4pt]
&\iff x\in\bigcap_{\alpha\in I}(X\setminus G_\alpha)
\end{aligned}
$$

# Замкнутость замыкания

$$
\Large
 E\subset X\implies\operatorname{cl}E\text{ замкнуто}
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
\text{Пусть }x\notin\operatorname{cl}E\\[4pt]
x\notin E\text{ и }x\text{ не является предельной точкой }E,\\
\text{поэтому }\exists r>0:\quad B(x,r)\cap E=\varnothing
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Для каждого }y\in B(x,r)\text{ по открытости шара}\\
\text{существует окрестность }V_y\subset B(x,r)\\[4pt]
V_y\cap E=\varnothing\implies y\notin E
\text{ и }y\text{ не предельная точка }E\\[4pt]
\implies y\notin\operatorname{cl}E
\end{gathered}
$$

$$
\Large
\begin{gathered}
B(x,r)\subset X\setminus\operatorname{cl}E\\[4pt]
\implies X\setminus\operatorname{cl}E\text{ открыто}\\[4pt]
\implies\operatorname{cl}E\text{ замкнуто}
\end{gathered}
$$

# Замыкание как пересечение замкнутых множеств

$$
\Large
\operatorname{cl}E =\bigcap_{\substack{F\text{ замкнуто}\\E\subset F}}F
$$

$$
\Large
\begin{gathered}
\text{То есть }\operatorname{cl}E\text{ — минимальное по включению}\\
\text{замкнутое множество, содержащее }E
\end{gathered}
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
\text{Пусть }F\text{ замкнуто и }E\subset F\\[4pt]
\text{Каждая предельная точка }E\text{ является предельной точкой }F,\\
\text{поэтому принадлежит }F.\quad\text{Значит, }\operatorname{cl}E\subset F
\end{gathered}
$$

$$
\Large
\operatorname{cl}E\subset\bigcap_{\substack{F\text{ замкнуто}\\E\subset F}}F
$$

$$
\Large
\begin{gathered}
\operatorname{cl}E\text{ замкнуто по пункту 1 и содержит }E,\\
\text{поэтому само входит в пересекаемое семейство}\\[4pt]
\bigcap_{\substack{F\text{ замкнуто}\\E\subset F}}F\subset\operatorname{cl}E
\end{gathered}
$$

# Внутренность как объединение открытых множеств

$$
\Large
\operatorname{int}E =\bigcup_{\substack{D\text{ открыто}\\D\subset E}}D
$$

$$
\Large
\begin{gathered}
\text{То есть }\operatorname{int}E\text{ — максимальное по включению}\\
\text{открытое множество, содержащееся в }E
\end{gathered}
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
\text{Если }D\text{ открыто и }D\subset E,\text{ то каждая точка }D\\
\text{имеет шар, лежащий в }D\text{ и, следовательно, в }E\\[4pt]
D\subset\operatorname{int}E\\[4pt]
\implies\bigcup_{\substack{D\text{ открыто}\\D\subset E}}D\subset\operatorname{int}E
\end{gathered}
$$

$$
\Large
\begin{gathered}
\operatorname{int}E\text{ открыто и }\operatorname{int}E\subset E,\\
\text{поэтому входит в объединяемое семейство}\\[4pt]
\operatorname{int}E\subset\bigcup_{\substack{D\text{ открыто}\\D\subset E}}D
\end{gathered}
$$

# Определение. Граничная точка и граница множества

$$
\Large
\begin{gathered}
 p\text{ — граничная точка }E,\text{ если для каждой }V_p\\[4pt]
V_p\cap E\ne\varnothing,\qquad
V_p\cap(X\setminus E)\ne\varnothing
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Множество граничных точек называется границей }E\\
\text{и обозначается }\partial E\\[4pt]
\partial E=\{p:\ \forall V_p,\ V_p\cap E\ne\varnothing
\text{ и }V_p\cap(X\setminus E)\ne\varnothing\}
\end{gathered}
$$

---

# Замыкание через границу

$$
\Large
\operatorname{cl}E=E\cup\partial E
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
\text{Если }p\notin E,\text{ то}\\[4pt]
p\in\operatorname{cl}E \iff p\text{ — предельная точка }E\\[4pt]
\iff\forall V_p:\ V_p\cap E\ne\varnothing
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{При этом каждая }V_p\text{ содержит }p\in X\setminus E,\\
\text{поэтому }V_p\cap(X\setminus E)\ne\varnothing\\[4pt]
p\in\operatorname{cl}E\setminus E\iff p\in\partial E\setminus E
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Точки }E\text{ принадлежат обеим частям равенства, поэтому}\\[4pt]
\operatorname{cl}E=E\cup\partial E
\end{gathered}
$$

---

#  Граница через замыкание и внутренность

$$
\Large
\partial E=\operatorname{cl}E\setminus\operatorname{int}E
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
p\in\operatorname{cl}E \iff\forall V_p:\ V_p\cap E\ne\varnothing,\\[6pt]
p\notin\operatorname{int}E \iff\forall V_p:\ V_p\cap(X\setminus E)\ne\varnothing
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Вместе эти условия означают }p\in\partial E\\[4pt]
\partial E=\operatorname{cl}E\setminus\operatorname{int}E
\end{gathered}
$$

---

# Замкнутость границы

$$
\Large
\partial E\text{ замкнута}
$$

### Доказательство (упражнение)

$$
\Large
\partial E=\operatorname{cl}E\cap(X\setminus\operatorname{int}E)
$$

$$
\Large
\begin{gathered}
\operatorname{cl}E\text{ замкнуто, а }\operatorname{int}E\text{ открыто,}\\
\text{поэтому }X\setminus\operatorname{int}E\text{ замкнуто}\\[4pt]
\text{Пересечение замкнутых множеств замкнуто,}\quad
\partial E\text{ замкнута}
\end{gathered}
$$

---

#  Критерий замкнутости через замыкание

$$
\Large
E\text{ замкнуто}\iff E=\operatorname{cl}E
$$

### Доказательство (упражнение)

$$
\Large
\begin{gathered}
\text{Если }E\text{ замкнуто, оно содержит все свои предельные точки,}\\
\text{поэтому }\operatorname{cl}E=E\\[6pt]
\text{Если }E=\operatorname{cl}E,\text{ то }E\text{ содержит все свои предельные точки}\\
\text{и по определению замкнуто}
\end{gathered}
$$
