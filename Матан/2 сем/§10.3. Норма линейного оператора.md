# Определение. Норма линейного оператора

$$
\Large
\text{Пусть }(X,\|\cdot\|_X),(Y,\|\cdot\|_Y)\text{ — нормированные пространства}
$$
$$\Large
L:X\to Y\text{ — линейный оператор}
$$

$$
\Large
\text{Нормой линейного оператора }L\text{ называется }\|L\|=\|L\|_{X\to Y}=\sup_{\|x\|_X\le1}\|Lx\|_Y
$$

---

# Определение. Ограниченный линейный оператор

$$
\Large
L:X\to Y\text{ ограничен, если }\|L\|<+\infty
$$

---

# Обозначение. Пространство ограниченных линейных операторов

$$
\Large
\mathcal L(X,Y)\text{ — пространство ограниченных линейных операторов }X\to Y
$$

$$
\Large
\text{Для операторов }X\to X\text{ используется обозначение }\mathcal L(X)
$$

---

# Лемма 2. Равные выражения для нормы линейного оператора

$$
\Large
\begin{aligned}
\|L\|&=\underbrace{\sup_{\|x\|\le1}\|Lx\|}_{N_1}=\underbrace{\sup_{\|x\|<1}\|Lx\|}_{N_2}=\underbrace{\sup_{\|x\|=1}\|Lx\|}_{N_3}\\
&=\underbrace{\sup_{x\ne\Theta_X}\frac{\|Lx\|}{\|x\|}}_{N_4}=\underbrace{\inf\{C>0:\ \forall x\in X\quad\|Lx\|\le C\|x\|\}}_{N_5}
\end{aligned}
$$

$$
\Large
\inf\varnothing=+\infty
$$

### Доказательство

$$
\Large
\text{Очевидны }N_2\le N_1,\qquad N_3\le N_1,\qquad N_3\le N_4
$$

$$
\Large
1)\quad N_1\le N_5
$$

$$
\Large
N_5=\inf\{C>0:\ \|Lx\|\le C\|x\|\ \forall x\in X\}\implies\forall\varepsilon>0\quad\|Lx\|\le(N_5+\varepsilon)\|x\|
$$

$$
\Large
\forall x\in X,\quad\|x\|\le1\implies\|Lx\|\le(N_5+\varepsilon)\|x\|\le N_5+\varepsilon
$$

$$
\Large
\implies N_1=\sup_{\|x\|\le1}\|Lx\|\le N_5+\varepsilon\implies N_1\le N_5
$$

$$
\Large
2)\quad N_5\le N_4
$$

$$
\Large
N_4=\sup_{x\ne\Theta_X}\frac{\|Lx\|}{\|x\|}\implies\|Lx\|\le N_4\|x\|\quad\forall x\ne\Theta_X\implies N_5\le N_4
$$

$$
\Large
\text{Для }N_5\text{ число }N_4\text{ — одно из значений }C
$$

$$
\Large
3)\quad N_4\le N_2
$$

$$
\Large
\|Lx\|\le N_2\qquad\forall x\in X:\quad\|x\|<1
$$

$$
\Large
\text{Пусть }\varepsilon>0,\quad x\ne\Theta_X,\qquad\widetilde x=\frac{x}{(1+\varepsilon)\|x\|}\implies\|\widetilde x\|=\frac1{1+\varepsilon}<1
$$

$$
\Large
\frac{\|Lx\|}{(1+\varepsilon)\|x\|}=\|L\widetilde x\|\le N_2\implies\frac{\|Lx\|}{\|x\|}\le N_2(1+\varepsilon)
$$

$$
\Large
\implies N_4=\sup_{x\ne\Theta_X}\frac{\|Lx\|}{\|x\|}\le(1+\varepsilon)N_2\implies N_4\le N_2
$$

$$
\Large
4)\quad N_2\le N_3,\qquad N_3=\sup_{\|x\|=1}\|Lx\|
$$

$$
\Large
\text{Пусть }x\in X:\quad\|x\|<1,\qquad\widetilde x=\frac{x}{\|x\|},\quad\|\widetilde x\|=1,\quad\|L\widetilde x\|\le N_3
$$

$$
\Large
\implies\|Lx\|\le N_3\|x\|\le N_3\implies N_2=\sup_{\|x\|<1}\|Lx\|\le N_3
$$

$$
\Large
N_3\le N_1\le N_5\le N_4\le N_2\le N_3
$$

# Следствие 1. Верхняя и нижняя оценки нормы оператора

$$
\Large
\text{Если }\exists M>0:\quad\forall x\in X\quad\|Lx\|_Y\le M\|x\|_X\implies L\in\mathcal L(X,Y)\text{ и }\|L\|\le M
$$

$$
\Large
\text{Если }\exists m>0,\ x_0\in X:\quad\|Lx_0\|_Y\ge m\|x_0\|\implies\|L\|\ge m
$$

---

# Следствие 2. Эквивалентность ограниченности и непрерывности линейного оператора

$$
\Large
X,Y\text{ — нормированные пространства},\qquad L:X\to Y\text{ — линейный оператор}
$$

$$
\Large
\text{Следующие условия эквивалентны}
$$

$$
\Large
\begin{gathered}
1)\quad L\text{ — ограниченный линейный оператор}\\
2)\quad L\text{ непрерывен в }\Theta_X\\
3)\quad L\text{ непрерывен на }X\\
4)\quad L\text{ равномерно непрерывен}\\
5)\quad E\text{ ограничено}\implies L(E)\text{ ограничено}\\
6)\quad L(B(\Theta_X,1))\text{ ограничено}
\end{gathered}
$$

### Доказательство

$$
\Large
1\Rightarrow2:\quad L\Theta_X=\Theta_Y,\qquad\lim_{x\to\Theta}\|Lx\|=0,\text{ так как }\|Lx\|\le\|L\|\|x\|
$$

$$
\Large
\implies Lx\xrightarrow[x\to\Theta_X]{}\Theta_Y=L\Theta_X\implies L\text{ непрерывен в нуле}
$$

$$
\Large
2\Rightarrow4:\quad x_0\in X,\quad Lx_0=y_0,\qquad\|Lx-Lx_0\|\le\|L\|\|x-x_0\|
$$

$$
\Large
\text{Пусть }\varepsilon>0,\quad\|x-x_0\|<\delta=\frac{\varepsilon}{\|L\|}\implies\|Lx-Lx_0\|\le\|L\|\|x-x_0\|<\varepsilon
$$

$$
\Large
4\Rightarrow3\text{ — очевидно},\qquad3\Rightarrow2\text{ — очевидно}
$$

$$
\Large
2\Rightarrow1:\quad\text{Пусть }L\text{ — линейный оператор, непрерывный в }\Theta_X
$$

$$
\Large
\varepsilon=1,\quad\exists\delta>0:\quad\|x\|=\|x-\Theta_X\|<\delta\implies\|Lx\|=\|Lx-L\Theta_X\|\le1
$$

$$
\Large
\text{Пусть }x\in X\setminus\{\Theta_X\},\qquad\widetilde x=\delta\frac{x}{\|x\|}\implies\|\widetilde x\|=\delta\implies\|L\widetilde x\|\le1
$$

$$
\Large
\|Lx\|\le\frac1\delta\|x\|\implies\|L\|\le\frac1\delta<+\infty\implies L\text{ ограничен}
$$

$$
\Large
5\Rightarrow6\text{ — очевидно}
$$

$$
\Large
1\iff6:\quad L\text{ ограничен}\iff\exists C>0:\quad\forall\|x\|<1\quad\|Lx\|\le C
$$

$$
\Large
\iff\exists C>0:\quad L(B(\Theta_X,1))\subset B(\Theta_Y,C)
$$

$$
\Large
1\Rightarrow5
$$

$$
\Large
\text{Пусть }E\text{ ограничено},\quad\exists R>0:\quad\forall x\in E\quad\|x\|\le R
$$

$$
\Large
\|Lx\|\le\|L\|\|x\|\le\|L\|R\implies Lx\in B(\Theta_Y,\|L\|R)
$$

$$
\Large
\implies LE\subset B(\Theta_Y,\|L\|R)\text{ — ограниченное множество}
$$

# Теорема. Оценка нормы оператора, порождённого матрицей

$$
\Large
\text{Пусть }L_A:\mathbb R^n\to\mathbb R^m\text{ — линейный оператор, порождённый матрицей }A=(a_{jk})_{\substack{j=1,\ldots,m\\k=1,\ldots,n}}
$$

$$
\Large
L_A\text{ ограничен},\qquad\|L_A\|^2\le\sum_{j=1}^m\sum_{k=1}^n a_{jk}^2
$$

### Доказательство

$$
\Large
\text{Пусть }x\in\mathbb R^n
$$

$$
\Large
\|L_Ax\|^2=\left\|\sum_{j=1}^m\left(\sum_{k=1}^n a_{jk}x_k\right)e^j\right\|^2=\sum_{j=1}^m\left(\sum_{k=1}^n a_{jk}x_k\right)^2\overset{\text{Коши — Буняковского — Шварца}}{\le}\left(\sum_{j=1}^m\sum_{k=1}^n a_{jk}^2\right)\|x\|^2
$$

---

### Пример 1. Норма линейного функционала на Rⁿ

$$
\Large
L_A:\mathbb R^n\to\mathbb R,\qquad A=(a_1,\ldots,a_n)
$$

$$
\Large
L_A(x)=(a_1,\ldots,a_n)\begin{pmatrix}x_1\\\vdots\\x_n\end{pmatrix}=\sum_{k=1}^n a_kx_k=\langle A,x\rangle
$$

### Доказательство

$$
\Large
L_A((a_1,\ldots,a_n))=\sum_{k=1}^n a_k^2\implies\|L_A\|\ge\frac{\sum_{k=1}^n a_k^2}{\sqrt{\sum_{k=1}^n a_k^2}}=\sqrt{\sum_{k=1}^n a_k^2}
$$

$$
\Large
\implies\|L_A\|=\sqrt{\sum_{k=1}^n a_k^2}
$$

$$
\Large
\text{Обратное неравенство }\le\text{ следует из теоремы 2}
$$

---

### Пример 2. Норма оператора из R в Rⁿ

$$
\Large
L_A:\mathbb R^1\to\mathbb R^n,\qquad A=\begin{pmatrix}a_1\\\vdots\\a_n\end{pmatrix}
$$

$$
\Large
L_Ax=\begin{pmatrix}a_1\\\vdots\\a_n\end{pmatrix}x=\begin{pmatrix}a_1x\\\vdots\\a_nx\end{pmatrix}
$$

### Доказательство

$$
\Large
x=1,\qquad\|L_A1\|=\|(a_1,\ldots,a_n)^T\|=\sqrt{\sum_{k=1}^n a_k^2}
$$

$$
\Large
\|L_A\|\ge\sqrt{\sum_{k=1}^n a_k^2},\qquad\le\text{ следует из теоремы 2}\implies\|L_A\|=\sqrt{\sum_{k=1}^n a_k^2}
$$

# Замечание 1. Произведение линейных операторов и оценка его нормы

$$
\Large
\text{Пусть }A:X\to Y,\quad B:Y\to Z\text{ — линейные операторы}
$$

$$
\Large
\text{Произведение операторов }BA\text{ — композиция }B\circ A,\qquad BA(x)=B(A(x))
$$

$$
\Large
\|BA\|\le\|B\|\cdot\|A\|
$$

### Доказательство

$$
\Large
\|BA(x)\|\le\|B\|\|Ax\|\le\|B\|\|A\|\|x\|
$$

---

# Замечание 2. Произведение операторов, порождённых матрицами

$$
\Large
A,B\text{ — матрицы},\qquad L_BL_A=L_{BA}
$$

---

# Замечание 3. Обратный оператор, порождённый матрицей

$$
\Large
A\text{ — матрица размера }n\times n,\quad\det A\ne0,\qquad\exists L_A^{-1}=L_{A^{-1}}
$$
