# Определение. Покрытие множества

$$
\Large
(X,d)\text{ — метрическое пространство},\qquad K\subset X
$$

$$
\Large
\begin{gathered}
\text{Семейство }\{G_\alpha\}_{\alpha\in I}\text{ — покрытие }K,\text{ если}\\[4pt]
K\subset\bigcup_{\alpha\in I}G_\alpha
\end{gathered}
$$

---

# Определение. Открытое покрытие

$$
\Large
\begin{gathered}
\text{Покрытие }\{G_\alpha\}_{\alpha\in I}\text{ открыто,}\\
\text{если все }G_\alpha\text{ открыты}
\end{gathered}
$$

---

# Определение. Подпокрытие

$$
\Large
\text{Пусть }I_1\subset I\text{ и }K\subset\bigcup_{\alpha\in I_1}G_\alpha
$$

$$
\Large
\implies\{G_\alpha\}_{\alpha\in I_1}\text{ — подпокрытие}
$$

---

# Определение. Компактное множество

$$
\Large
\begin{gathered}
K\subset X\text{ — компакт, если каждое открытое покрытие }K\\
\text{содержит конечное подпокрытие}
\end{gathered}
$$

---

### Пример 1

$$
\Large
[a;b]\text{ — компакт в }\mathbb{R}
$$

---

### Пример 2

$$
\Large
\{a\}\subset X\text{ — компакт}
$$

# Теорема. Замкнутость и ограниченность компакта

$$
\Large
(X,d)\text{ — метрическое пространство},\qquad K\subset X\text{ — компакт}
$$

$$
\Large
\implies K\text{ замкнут и ограничен}
$$

### Доказательство

$$
\Large
1)\quad\text{Пусть }p\text{ — предельная точка }K\text{ и }p\notin K
$$

$$
\Large
K\subset\bigcup_{n=1}^{\infty}
\left(X\setminus\overline{B}\left(p,\frac{1}{n}\right)\right) =X\setminus\{p\}
$$

$$
\Large
\begin{gathered}
G_n=X\setminus\overline{B}\left(p,\frac{1}{n}\right)\text{ открыто},\\[4pt]
\{G_n\}_{n=1}^{\infty}\text{ — покрытие }K\\[4pt]
\implies\{G_n\}_{n=1}^{\infty}\text{ содержит конечное подпокрытие}
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{Так как }G_1\subset G_2\subset\cdots\subset G_n\subset G_{n+1}\subset\cdots,\\[4pt]
\text{то }\exists N\in\mathbb{N}:\quad K\subset G_N\\[4pt]
\text{(}N\text{ — максимальный номер множества, входящего в подпокрытие)}
\end{gathered}
$$

$$
\Large
\begin{gathered}
\implies K\subset X\setminus\overline{B}\left(p,\frac{1}{N}\right)\\[4pt]
\implies B\left(p,\frac{1}{N}\right)\cap K=\varnothing\\[4pt]
\implies p\text{ не является предельной точкой }K\\[4pt]
\implies p\in K\implies K\text{ замкнут}
\end{gathered}
$$

$$
\Large
2)\quad\text{Пусть }p\in X\implies K\subset X=\bigcup_{n=1}^{\infty}B(p,n)
$$

$$
\Large
\implies\exists N:\quad K\subset B(p,N)\implies K\text{ ограничен}
$$

# Теорема. Замкнутое подмножество компакта

$$
\Large
\begin{gathered}
K\subset X\text{ — компакт},\qquad F\subset K\text{ замкнуто}\\[4pt]
\implies F\text{ — компакт}
\end{gathered}
$$

### Доказательство

$$
\Large
\text{Пусть }\{G_\alpha\}_{\alpha\in I}\text{ — открытое покрытие }F
$$

$$
\Large
\begin{gathered}
\implies K\subset\left(\bigcup_{\alpha\in I}G_\alpha\right)\cup(X\setminus F)\\[4pt]
\implies\{G_\alpha\}_{\alpha\in I}\cup\{X\setminus F\}\\
\text{— открытое покрытие }K
\end{gathered}
$$

$$
\Large
\begin{gathered}
\implies\exists I_1\subset I\text{ — конечное, такое, что}\\[4pt]
\{G_\alpha\}_{\alpha\in I_1}\cup\{X\setminus F\}\\
\text{— конечное открытое покрытие }K
\end{gathered}
$$

$$
\Large
\begin{gathered}
\implies\{G_\alpha\}_{\alpha\in I_1}\text{ — конечное открытое покрытие }F\\[4pt]
\implies F\text{ — компакт}
\end{gathered}
$$

# Определение. Секвенциальная компактность

$$
\Large
(X,d)\text{ — метрическое пространство}
$$

$$
\Large
\begin{gathered}
K\subset X\text{ секвенциально компактен, если}\\[4pt]
\forall\{x_n\}\subset K\quad\exists\{n_j\}\uparrow\uparrow:\quad
\{x_{n_j}\}\text{ сходится к элементу }K
\end{gathered}
$$

---

# Теорема. Эквивалентность компактности и секвенциальной компактности

$$
\Large
(X,d)\text{ — метрическое пространство}
$$

$$
\Large
K\subset X\text{ — компакт}\iff K\text{ секвенциально компактен}
$$

---

# Теорема. Критерий компактности в $\mathbb{R}^n$

$$
\Large
K\subset\mathbb{R}^n\implies\text{следующие условия эквивалентны:}
$$

$$
\Large
\begin{gathered}
1)\quad K\text{ — компакт}\\[6pt]
2)\quad K\text{ секвенциально компактен}\\[6pt]
3)\quad K\text{ ограничен и замкнут}
\end{gathered}
$$

### Доказательство

$$
\Large
1\iff2\quad\text{(из предыдущей теоремы)}
$$

$$
\Large
\begin{gathered}
1\implies3\text{ доказали в теореме о свойствах компакта}\\
\text{в любом метрическом пространстве, в частности в }\mathbb{R}^n
\end{gathered}
$$

$$
\Large
\begin{gathered}
3\implies1:\quad\text{Пусть }K\text{ ограничен и замкнут}\\[4pt]
\implies\exists\text{ замкнутый шар }\overline{B}\text{ такой, что}\\[4pt]
K\subset\overline{B}\subset[a,b]^n
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{(Любой шар в }\mathbb{R}^n\text{ — подмножество}\\
\text{некоторого замкнутого куба)}
\end{gathered}
$$

$$
\Large
\begin{gathered}
\text{При этом }[a,b]^n\text{ — компакт}\\
\text{(доказательство в следующей лемме)}\\[4pt]
\implies K\text{ — замкнутое подмножество компакта}\\[4pt]
\implies K\text{ — компакт}
\end{gathered}
$$

# Замечание. Покоординатная сходимость

$$
\Large
\{x^{(k)}\}_{k=1}^{\infty}\subset\mathbb{R}^n\text{ сходится к }a\in\mathbb{R}^n,\text{ то есть}
$$

$$
\Large
(x_1^{(k)},x_2^{(k)},\ldots,x_n^{(k)})\xrightarrow[k\to\infty]{}(a_1,\ldots,a_n)
\iff\forall j=1,\ldots,n:\quad x_j^{(k)}\xrightarrow[k\to\infty]{}a_j
$$

# Лемма 1. Компактность замкнутого куба

в конспекте вместо $k_i$ бралось $k_0$ - вопрос

$$
\Large
[a,b]^n\text{ (секвенциально) компактно}
$$

### Доказательство

$$
\Large
n=1:\quad\text{Пусть }\{x^{(k)}\}\subset[a,b]^n\text{ — ограниченная последовательность}
$$

$$
\Large
\begin{gathered}
\text{Это про то, что из любой ограниченной последовательности}\\
\text{можно выбрать сходящуюся подпоследовательность}
\end{gathered}
$$

$$
\Large
\implies\exists\{k_i\}_{i=1}^{\infty}\uparrow\uparrow:\quad
x^{(k_i)}\xrightarrow[i\to\infty]{}x\in\mathbb{R}
$$

$$
\Large
\text{При этом, так как }a\le x^{(k)}\le b,\text{ то }a\le x=\lim_{i\to\infty}x^{(k_i)}\le b
$$

$$
\Large
\text{Индукционный переход }n\to n+1:\quad\text{Пусть }x^{(k)}\in[a,b]^{n+1}
$$

$$
\Large
\widetilde{x}^{(k)}=(x_1^{(k)},\ldots,x_n^{(k)})\in[a,b]^n
\implies\text{по индукционному предположению}
$$

$$
\Large
\exists\{k_i\}_{i=1}^{\infty}\uparrow\uparrow:\quad
\widetilde{x}^{(k_i)}\xrightarrow[i\to\infty]{}\widetilde{x}\in[a,b]^n
$$

$$
\Large
\begin{gathered}
\text{Последовательность }\{x_{n+1}^{(k_i)}\}_{i=1}^{\infty}\\
\text{содержит сходящуюся подпоследовательность (по базе индукции)}
\end{gathered}
$$

$$
\Large
\implies\exists\{i_j\}_{j=1}^{\infty}\uparrow\uparrow:\quad
x_{n+1}^{(k_{i_j})}\xrightarrow[j\to\infty]{}\widehat{x}
$$

$$
\Large
\implies x^{(k_{i_j})}\xrightarrow[j\to\infty]{}
\left(\underbrace{\widetilde{x}}_{\in[a,b]^n},\underbrace{\widehat{x}}_{\in[a,b]^1}\right)
\in[a,b]^{n+1}
$$
