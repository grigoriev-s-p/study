$$
---
$$\Large
|x|<1 \qquad r \notin \mathbb{N} \cup \{0\}
$$
$$\Large
((1+x)^r)' = r(1+x)^{r-1}
$$
$$\Large
((1+x)^r)^{(n+1)} = r(r-1)\dots (r-n)(1+x)^{r-n-1}
$$
$$\Large
\left. ((1+x)^r)^{(k)}\right|_{x=0} = r(r-1)\dots (r-k+1)
$$
$$\Large
\text{Воспользуемся формулой Тейлора с остатком в интегральной форме:}
$$
$$\Large
(1+x)^r = 1 + \frac{rx}{1!} + \frac{r(r-1)}{2!}x^2 + \dots + \frac{r(r-1)\dots(r-n+1)}{n!} x^n +
$$
$$\Large
+ \frac{r(r-1)\dots(r-n)}{n!} \int\limits_0^x (x-t)^n (1+t)^{r-n-1} dt 
$$
$$\Large
(x-t)^n (1+t)^{r-n-1} = \frac{(x-t)^n}{(1+t)^n} \cdot (1+t)^{r-1}
$$
$$\Large
\left| \frac{x-t}{1+t} \right| \le |x| \qquad (tx>0, \quad|t|<|x|)
$$
$$\Large
(1+t)^{r-1} \le \begin{cases}  
2^{r-1}, \ \text{ если $r>1$}\\  
\frac{1}{(1-|x|)^{1-r}}, \ \text{ если $r<1$}
\end{cases} = C_0
$$
$$\Large
\Rightarrow \left| \frac{r(r-1)\dots(r-n)}{n!} \int\limits_0^x (x-t)^n (1+t)^{r-n-1} dt  \right| \le
$$
$$\Large
\le C_0 \frac{|r(r-1)\dots(r-n)|}{n!} |x|^n \int\limits_0^x dt = C_0 \frac{|r(r-1)\dots(r-n)|}{n!} |x|^{n+1}
$$
$$\Large
P_n = \frac{|r(r-1)\dots(r-n)|}{n!} |x|^{n+1}
$$
$$\Large
n+1>r
$$
$$\Large
\frac{P_{n+1}}{P_n} = \frac{|r-n-1|}{n+1} |x| = \frac{n+1-r}{n+1} |x| = (1- \frac{r}{n+1})|x| \le (1 + \frac{|r|}{n+1})|x|
$$
$$\Large
\frac{|r|}{n+1} \xrightarrow[n\to \infty]{} 0 \qquad \text{Выберем $n_1$, т.ч. при $n>n_1$}:
$$
$$\Large
1+ \frac{r}{n+1} < \frac{1}{\sqrt{|x|}}
$$
$$\Large
\Rightarrow \ \text{при $n>n_1$} \quad \frac{P_{n+2}}{P_{n+1}} < |x| \cdot \frac{1}{\sqrt{|x|}} = \sqrt{|x|}
$$
$$\Large
P_n = \frac{P_n}{P_{n-1}} \cdot \frac{P_{n-1}}{P_{n-2}} \dots \frac{P_{n_1 +1}}{P_{n_1}} \cdot P_{n_1} < (\sqrt{|x|})^{n-n_1} \cdot P_{n_1} \xrightarrow[n\to \infty]{} 0
$$
$$\Large
\text{Получаем } \quad (1+x)^r = 1 +rx + \sum_{k=2}^{\infty} \frac{r(r-1)\dots(r-k+1)}{k!} x^k
$$
