$$\Large
\text{В любом разложении мы берем степенной ряд с $a = 0$}
$$
$$\Large
\text{Основная суть дальнейших действий - избавиться от последнего слагаемого}
$$
---
$$\Large
x \ne 0, \qquad \exists c, \quad cx>0, \quad |c|<|x|
 $$
$$ \Large
\text{По формуле Тейлора с ост. в форме Лагранжа(1.80):}
$$
$$ \Large
e^x = 1 + \sum_{k=1}^{n} \frac{x^k}{k!} + \frac{e^c}{(n+1)!} x^{n+1}
$$
$$ \Large
\text{Выберем $n_0$, т.ч. } \quad n_0> |x| + 1, \quad n > n_0
$$
$$ \Large
\frac{|x|^{n+1}}{(n+1)!} =\frac{|x|^{n_0+1}}{(n_0+1)!} \cdot \frac{|x|^{n-n_0}}{(n_0+2) \dots (n+1)} =
$$
$$ \Large
=\frac{|x|^{n_0+1}}{(n_0+1)!} \cdot \frac{|x|}{n_0+2}\cdot \dots \cdot\frac{|x|}{n+1} < \frac{|x|^{n_0+1}}{(n_0+1)!} \cdot q^{n-n_0} \xrightarrow[n \to \infty]{} 0
$$
$$ \Large
\text{где } \quad q = \frac{|x|}{n_0 + 2} < 1
$$
$$ \Large
e^c < e^{|x|}
$$
$$ \Large
\text{Получаем: }\quad \frac{e^c}{(n+1)!} x^{n+1} \xrightarrow[n \to \infty]{} 0
$$
$$ \Large
\Rightarrow e^x = 1 + \sum_{n=1}^{\infty} \frac{x^n}{n!}, \quad x\in \mathbb{R}
$$
$$ \Large
x = 1: \quad e = 2 + \sum_{n=2}^{\infty} \frac{1}{n!}
$$
