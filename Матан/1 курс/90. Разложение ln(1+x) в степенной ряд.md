$$\Large
\text{В любом разложении мы берем степенной ряд с $a = 0$}
$$
$$\Large
\text{Основная суть дальнейших действий - избавиться от последнего слагаемого}
$$
---
$$\Large
|x|<1
$$
$$\Large
(\ln(1+x))' = \frac{1}{1+x}
$$
$$\Large
(\ln(1+x))^{(n+1)} = (-1)^n \cdot \frac{n!}{(1+x)^{n+1}}
$$
$$\Large
\left. (\ln(1+x))' \right|_{x=0} = 1
$$
$$\Large
\left. (\ln(1+x))^{(n+1)} \right|_{x=0} = (-1)^n \cdot n!
$$
$$\Large
\text{Воспользуемся формулой Тейлора с остатком в интегральной форме:}
$$
$$\Large
\ln(1+x) = \sum_{k=1}^n  \frac{(-1)^{k-1}x^k}{k} + \frac{1}{n!} \cdot n! \cdot \int\limits_0^x (x-t)^n \frac{(-1)^n}{(1+t)^{n+1}}dt =
$$
$$\Large
=\sum_{k=1}^n  \frac{(-1)^{k-1}x^k}{k} + (-1)^n \int\limits_0^x \frac{(x-t)^n}{(1+t)^n} \frac{1}{(1+t)}dt 
$$
$$\Large
\text{Рассмотрим 2 случая: }\ 0<x<1 \ \ \text{ и } \ -1<x<0 
$$
1)
$$\Large
0<x<1 \qquad 0<t<x 
$$
$$\Large
\frac{x-t}{1+t} < x \ \Rightarrow \ \left|(-1)^n  \int\limits_0^x \frac{(x-t)^n}{(1+t)^n}\frac{dt}{1+t} \right| <
$$
$$\Large
< x^n \cdot \int\limits_0^x dt = x^{n+1} \xrightarrow[n\to \infty]{} 0
$$

$$\Large
\text{Получаем: } \ln(1+x) = \sum_{k=1}^{\infty} (-1)^{k-1} \frac{x^k}{k}
$$
2)
$$\Large
-1<x<0 \qquad x<t<0
$$
$$\Large
1+t > 1 - |x| \ \Rightarrow \ \frac{1}{1+t}<\frac{1}{1-|x|} 
$$
$$\Large
\left| \frac{x-t}{1+t} \right| =\left| \frac{-|x|+|t|}{1-|t|} \right| = \left| -\frac{|x|-|t|}{1-|t|} \right| \le |x|
$$
$$\Large
\Rightarrow \left|(-1)^n  \int\limits_0^x \frac{(x-t)^n}{(1+t)^n}\frac{dt}{1+t} \right| \le \frac{|x|^{n+1}}{1-|x|} \xrightarrow[n \to \infty]{} 0
$$
$$\Large
\text{То есть для } |x|<1: \quad \ln(1+x) =  \sum_{k=1}^{\infty} (-1)^{k-1} \frac{x^k}{k}
$$
