$$
\Large
\int\limits_{1}^{\infty}\frac{\sin x}{x}\,dx
$$
$$\Large
\text{Воспользуемся признаком Дирихле}
$$
$$
\Large
f(x)=\sin x,
\qquad
g(x)=\frac{1}{x}
$$

$$
\Large
\left|
\int\limits_{1}^{b}f(x)\, dx
\right|
=
\left|
\int\limits_{1}^{b}\sin x\,dx
\right|
=
\left|
-\cos b+\cos 1
\right|
\le 2
$$
$$\Large
g(x) \xrightarrow[x \to \infty]{} 0
$$
$$
\Large
\Rightarrow
\text{исходный интеграл сходится}
$$

## Хотим доказать что сходится неабсолютно 
$$
\Large
\int\limits_{1}^{\infty}\left|\frac{\sin x}{x}\right|\,dx
\quad \text{расходится}
$$

$$
\Large
|\sin x|\ge \sin^2 x
$$

$$
\Large
\frac{|\sin x|}{x}\ge \frac{\sin^2 x}{x}
$$

$$
\Large
\int\limits_{1}^{\infty}\frac{\sin^2 x}{x}\,dx
\quad \text{расходится }?
$$

$$
\Large
\sin^2 x
=
\frac{1-\cos 2x}{2}
=
\frac12-\frac{\cos 2x}{2}
$$

$$
\Large
\int\limits_{1}^{\infty}\frac{\sin^2 x}{x}\,dx
=
\int\limits_{1}^{\infty}
\left(
\frac{1}{2x}
-
\frac{\cos 2x}{2x}
\right)\,dx
 = \int\limits_{1}^{\infty}
\frac{1}{2x}dx + \int\limits_{1}^{\infty}\frac{\cos 2x}{2x} dx
$$

$$
\Large
\int\limits_{1}^{\infty}\frac{\cos 2x}{x}\,dx
\quad
\text{сходится }
(\text{по признаку Дирихле})
$$

$$
\Large
\int\limits_{1}^{\infty}\frac{dx}{2x}
\quad
\text{расходится}
\quad
(\text{как эталонный интеграл})
$$

$$
\Large
\Rightarrow
\int\limits_{1}^{\infty}\frac{\sin^2 x}{x}\,dx
\quad
\text{расходится}
$$

$$
\Large
\frac{|\sin x|}{x}\ge \frac{\sin^2 x}{x}
$$

$$
\Large
\Rightarrow
\int\limits_{1}^{\infty}\left|\frac{\sin x}{x}\right|\,dx
\quad
\text{расходится}
$$

$$
\Large
\Rightarrow
\int\limits_{1}^{\infty}\frac{\sin x}{x}\,dx
\quad
\text{сходится неабсолютно}
$$
