\sum_{n=1}^{\infty} n! x^n
$$
$$ \Large
b_n = n!
$$
$$ \Large
\frac{b_n}{b_{n+1}} = \frac{n!}{(n+1)!} = \frac{1}{n+1} \xrightarrow[n \to \infty]{} 0 \quad \Rightarrow \quad R=0
$$

2)
$$\Large
\sum_{n=1}^{\infty} \frac{x^n}{n!}
$$
$$ \Large
\frac{b_n}{b_{n+1}} = \frac{1}{n!} \cdot (n+1)! = n+1 \xrightarrow[n \to \infty]{}+ \infty \quad  \Rightarrow \quad R = +\infty
$$
3)
$$\Large
\sum_{n=1}^{\infty} \frac{x^n}{n}
$$
$$\Large
\frac{b_n}{b_{n+1}} = \frac{1}{n}\cdot (n+1) = 1 + \frac{1}{n} \xrightarrow[n\to \infty]{} 1 \quad \Rightarrow \quad R=1
$$
